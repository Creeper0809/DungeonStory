            if (!hasWork)

                    "WIM017053_PREPARATION_BLOCKER_OWNER_RESEARCH_CELL_NOT_INTERACTABLE;actor="

            throw new InvalidOperationException(
                "WIM017053_PREPARATION_BLOCKER_NO_ACTIVE_RESEARCH_UI_ACTOR.");
        }
        return eligible.OrderBy(ActorId, StringComparer.Ordinal).ToArray();

            Canvas.ForceUpdateCanvases();
        }
        string selectedAfter = WorkPriority(workerWork, ResearchWorkTypeId());
        priorityTransitions.Add(selectedActorId + ":Research=" + selectedBefore
            + "->" + selectedAfter + "/clicks=" + selectedClicks);
        if (!string.Equals(selectedAfter, "Priority1", StringComparison.Ordinal))
        {
            throw new InvalidOperationException(
                "WIM017053_PREPARATION_BLOCKER_SELECTED_RESEARCH_PRIORITY1_UI_FAILED.");
        }

        foreach (Component actor in actors.Where(value => !ReferenceEquals(value, worker)))
        {
            if (!TryGetPublicWork(actor, out object work) || !ResearchEnabled(work))
            {
                continue;
            }
            string cellName = "Cell_" + actor.GetInstanceID() + "_Research";
            string before = WorkPriority(work, ResearchWorkTypeId());
            int clicks = 0;
            while (ResearchEnabled(work) && clicks < 4)
            {
                Click(FindActiveButton(cellName),
                    "staff-other-research;actor=" + ActorId(actor));
                clicks++;
                yield return null;
                Canvas.ForceUpdateCanvases();
            }
            string after = WorkPriority(work, ResearchWorkTypeId());
            priorityTransitions.Add(ActorId(actor) + ":Research=" + before
                + "->" + after + "/clicks=" + clicks);
            if (ResearchEnabled(work))
            {
                throw new InvalidOperationException(
                    "WIM017053_PREPARATION_BLOCKER_OTHER_RESEARCH_PRIORITY_OFF_UI_FAILED;actor="
                    + ActorId(actor));
            }
        }

        string[] enabled = actors
            .Where(value => TryGetPublicWork(value, out object work) && ResearchEnabled(work))
            .Select(ActorId)
            .OrderBy(value => value, StringComparer.Ordinal)
            .ToArray();
        if (enabled.Length != 1 || !string.Equals(enabled[0], selectedActorId, StringComparison.Ordinal))
        {
            throw new InvalidOperationException(
                "WIM017053_PREPARATION_BLOCKER_RESEARCH_UI_NOT_ISOLATED;enabled="
                + string.Join(",", enabled));
        }

        string researchId = IdValue(ResearchWorkTypeId());
        foreach (object definition in AllStaffPriorityWorkDefinitions())
        {
            object workTypeId = P(definition, "WorkTypeId");
            string workTypeText = IdValue(workTypeId);
            if (string.Equals(workTypeText, researchId, StringComparison.Ordinal))
            {
                continue;
            }
            string before = WorkPriority(workerWork, workTypeId);
            int clicks = 0;
            while (!string.Equals(WorkPriority(workerWork, workTypeId), "Off", StringComparison.Ordinal)
                && clicks < 4)
            {
                string cellName = "Cell_" + ((Component)worker).GetInstanceID()
                    + "_" + StaffCellSuffix(definition);
                Click(FindActiveButton(cellName), "staff-selected-nonresearch;actor="
                    + selectedActorId + ";work=" + workTypeText);
                clicks++;
                yield return null;
                Canvas.ForceUpdateCanvases();
            }
            string after = WorkPriority(workerWork, workTypeId);
            priorityTransitions.Add(selectedActorId + ":" + workTypeText + "="
                + before + "->" + after + "/clicks=" + clicks);
            if (!string.Equals(after, "Off", StringComparison.Ordinal))
            {
                throw new InvalidOperationException(
                    "WIM017053_PREPARATION_BLOCKER_SELECTED_NON_RESEARCH_UI_OFF_FAILED;work="
                    + workTypeText);
            }
        }
        if (!ResearchEnabled(workerWork)
            || !string.Equals(WorkPriority(workerWork, ResearchWorkTypeId()), "Priority1", StringComparison.Ordinal))
        {
            throw new InvalidOperationException(
                "WIM017053_PREPARATION_BLOCKER_SELECTED_RESEARCH_LOST_DURING_ISOLATION.");
        }
        string[] haulers = actors
            .Where(value => !ReferenceEquals(value, worker))
            .Where(value => TryGetPublicWork(value, out object work)
                && !string.Equals(WorkPriority(work, HaulWorkTypeId()), "Off", StringComparison.Ordinal))
            .Select(ActorId)
            .OrderBy(value => value, StringComparer.Ordinal)
            .ToArray();
        otherEnabledHaulers = haulers.Length == 0 ? "<none>" : string.Join(",", haulers);
        if (haulers.Length == 0)
        {
            throw new InvalidOperationException(
                "WIM017053_PREPARATION_BLOCKER_NO_OTHER_ACTOR_WITH_HAUL_ENABLED.");
        }
    }

    private IEnumerator WarmResearchFacilityCandidateIndex()
    {
        object grid = RequireLiveGrid();
        object cache = Resolve("IFacilityCandidateCache", "Assembly-CSharp");
        Type gridType = T("Grid", "DungeonStory.Grid");
        Type facilityWorkType = T("FacilityWorkType", "DungeonStory.Buildings");
        var getCandidates = cache.GetType().GetMethod("GetWorkCandidates", new[]
        {
            gridType,
            facilityWorkType
        });
        if (getCandidates == null || !getCandidates.IsPublic)
        {
            throw new InvalidOperationException(
                "Public IFacilityCandidateCache.GetWorkCandidates is unavailable.");
        }
        object research = Enum.Parse(facilityWorkType, "Research");
        int queries = 0;
        getCandidates.Invoke(cache, new[] { grid, research });
        queries++;
        bool pending = (bool)P(cache, "HasPendingIndexBuild");
        int versionBefore = Convert.ToInt32(P(cache, "CandidateIndexVersion"));
        if (!pending)
        {
            researchIndexWarmup = "pendingBefore=false;queries=" + queries
                + ";indexVersion=" + versionBefore + ";normalFrames=0";
            yield break;
        }

        float gameStart = Convert.ToSingle(P(clock, "Time"));
        float wallStart = Time.realtimeSinceStartup;
        int normalFrames = 0;
        try
        {
            Time.timeScale = 1f;
            yield return null;
            normalFrames++;
            if ((bool)P(clock, "IsPaused"))
            {
                throw new InvalidOperationException(
                    "WIM017053_PREPARATION_BLOCKER_RESEARCH_INDEX_SCHEDULER_PAUSED.");
            }
            while ((bool)P(cache, "HasPendingIndexBuild"))
            {
                float elapsedGame = Convert.ToSingle(P(clock, "Time")) - gameStart;
                float elapsedWall = Time.realtimeSinceStartup - wallStart;
                if (elapsedGame >= IndexWarmupMaximumGameSeconds
                    || elapsedWall >= IndexWarmupMaximumWallSeconds)
                {
                    throw new InvalidOperationException(
                        "WIM017053_PREPARATION_BLOCKER_RESEARCH_INDEX_WARMUP_TIMEOUT;game="
                        + elapsedGame + ";wall=" + elapsedWall);
                }
                getCandidates.Invoke(cache, new[] { grid, research });
                queries++;
                yield return null;
                normalFrames++;
            }
        }
        finally
        {
            Time.timeScale = 0f;
        }
        getCandidates.Invoke(cache, new[] { grid, research });
        queries++;
        if ((bool)P(cache, "HasPendingIndexBuild"))
        {
            throw new InvalidOperationException(
                "WIM017053_PREPARATION_BLOCKER_RESEARCH_INDEX_STILL_PENDING.");
        }
        researchIndexWarmup = "pendingBefore=true;queries=" + queries
            + ";normalFrames=" + normalFrames + ";game=" + gameStart + "->"
            + Convert.ToSingle(P(clock, "Time")) + ";indexVersion="
            + Convert.ToString(P(cache, "CandidateIndexVersion"));
    }

    private static bool TryFindEligibleQ03Position(
        object grid,
        object q03,
        out Vector2Int chosen)
    {
        chosen = default;
        object rooms = Resolve("IRoomLayoutCache", "Assembly-CSharp");
        Type gridType = T("Grid", "DungeonStory.Grid");
        Type roomType = T("RoomInstance", "Assembly-CSharp");
        Type gridLayer = T("GridLayer", "DungeonStory.Grid");
        var tryGetRoom = rooms.GetType().GetMethod("TryGetRoom", new[]
        {
            gridType,
            typeof(Vector2Int),
            roomType.MakeByRefType()
        });
        var isEligibleRoom = T("ResearchBlueprintArchiveDestinationAuthority", "Assembly-CSharp")
            .GetMethod("IsEligibleRoom", new[] { roomType });
        var getCells = grid.GetType().GetMethod("GetCells", Type.EmptyTypes);
        var getGridCell = grid.GetType().GetMethod("GetGridCell", new[] { typeof(Vector2Int) });
        var footprint = q03.GetType().GetMethod("GetGridPosList", new[] { typeof(Vector2Int) });
        if (tryGetRoom == null || !tryGetRoom.IsPublic
            || isEligibleRoom == null || !isEligibleRoom.IsPublic || !isEligibleRoom.IsStatic
            || getCells == null || !getCells.IsPublic
            || getGridCell == null || !getGridCell.IsPublic
            || footprint == null || !footprint.IsPublic)
        {
            throw new InvalidOperationException(
                "Public Q03 initial-placement room/grid query is unavailable.");
        }

        Vector2Int[] candidates = ((IEnumerable)getCells.Invoke(grid, null)).Cast<object>()
            .Select(cell => (Vector2Int)P(cell, "Position"))
            .OrderBy(position => position.y)
            .ThenBy(position => position.x)
            .ToArray();
        foreach (Vector2Int position in candidates)
        {
            IEnumerable cells = footprint.Invoke(q03, new object[] { position }) as IEnumerable;
            if (cells == null)
            {
                continue;
            }
            object expectedRoom = null;
            bool eligible = true;
            foreach (object rawCell in cells)
            {
                Vector2Int cellPosition = (Vector2Int)rawCell;
                object cell = getGridCell.Invoke(grid, new object[] { cellPosition });
                var canOccupy = cell?.GetType().GetMethod("CanOccupy", new[] { gridLayer });
                if (cell == null || canOccupy == null || !canOccupy.IsPublic
                    || !(bool)canOccupy.Invoke(cell, new[] { Enum.Parse(gridLayer, "Building") }))
                {
                    eligible = false;
                    break;
                }
                object[] roomArguments = { grid, cellPosition, null };
                if (!(bool)tryGetRoom.Invoke(rooms, roomArguments)
                    || roomArguments[2] == null
                    || !(bool)isEligibleRoom.Invoke(null, new[] { roomArguments[2] })
                    || (expectedRoom != null && !ReferenceEquals(expectedRoom, roomArguments[2])))
                {
                    eligible = false;
                    break;
                }
                expectedRoom = roomArguments[2];
            }
            if (eligible && expectedRoom != null)
            {
                chosen = position;
                return true;
            }
        }
        return false;
    }

    private IEnumerator EnsureArchive()
    {
        archiveQuery = Resolve("IResearchBlueprintArchiveQuery", "Assembly-CSharp");
        object[] existing = ValidArchives(archiveQuery);
        if (existing.Length > 0)
        {
            archiveDetail = "reused-authored-Q03;" + DescribeArchive(existing[0]);
            yield break;
        }

        object grid = RequireLiveGrid();
        object q03 = Resources.Load(Q03ResourcePath, T("BuildingSO", "Assembly-CSharp"));
        if (q03 == null || Convert.ToInt32(F(q03, "id")) != Q03DefinitionId)
        {
            throw new InvalidOperationException(
                "WIM017053_PREPARATION_BLOCKER_AUTHORED_Q03_DEFINITION_MISSING.");
        }
        if (!TryFindEligibleQ03Position(grid, q03, out Vector2Int position))
        {
            throw new InvalidOperationException(
                "WIM017053_PREPARATION_BLOCKER_ELIGIBLE_RESEARCH_ROOM_Q03_POSITION_MISSING.");
        }

        object controller = FindLoadedComponent(
            "DungeonStoryGridBuildingController", "Assembly-CSharp");
        Type initialBuildInfo = T("InitialBuildInfo", "Assembly-CSharp");
        object info = Activator.CreateInstance(initialBuildInfo);
        initialBuildInfo.GetField("Position")?.SetValue(info, position);
        initialBuildInfo.GetField("Building")?.SetValue(info, q03);
        if (initialBuildInfo.GetField("Position") == null || initialBuildInfo.GetField("Building") == null)
        {
            throw new InvalidOperationException("Public InitialBuildInfo fields are unavailable.");
        }
        Array placement = Array.CreateInstance(initialBuildInfo, 1);
        placement.SetValue(info, 0);
        Type enumerable = typeof(IEnumerable<>).MakeGenericType(initialBuildInfo);
        var place = controller.GetType().GetMethod("TryPlaceInitialBuildings", new[]
        {
            enumerable,
            typeof(string).MakeByRefType()
        });
        if (place == null || !place.IsPublic)
        {
            throw new InvalidOperationException(
                "Public DungeonStoryGridBuildingController.TryPlaceInitialBuildings is unavailable.");
        }
        object[] arguments = { placement, null };
        if (!(bool)place.Invoke(controller, arguments))
        {
            throw new InvalidOperationException(
                "WIM017053_PREPARATION_BLOCKER_INITIAL_Q03_PLACEMENT_REJECTED;message="
                + Convert.ToString(arguments[1]));
        }

        for (int frame = 0; frame < 6; frame++)
        {
            yield return null;
            existing = ValidArchives(archiveQuery);
            if (existing.Length > 0)
            {
                archiveDetail = "public-initial-Q03@" + position + ";message="
                    + Convert.ToString(arguments[1]) + ";" + DescribeArchive(existing[0]);
                yield break;
            }
        }
        throw new InvalidOperationException(
            "WIM017053_PREPARATION_BLOCKER_INITIAL_Q03_DID_NOT_JOIN_VALID_ARCHIVES;message="
            + Convert.ToString(arguments[1]));
    }

    private static int AvailableStock(object itemRuntime, string categoryName, out string detail)
    {
        Type category = T("StockCategory", "DungeonStory.Buildings");
        Type candidate = T("WorldItemStockCandidate", "Assembly-CSharp");
        Type candidateList = typeof(List<>).MakeGenericType(candidate);
        var copy = itemRuntime.GetType().GetMethod("CopyAvailableStockCandidates", new[]
        {
            category,
            candidateList
        });
        if (copy == null || !copy.IsPublic)
        {
            throw new InvalidOperationException(
                "Public IWorldItemStackRuntime.CopyAvailableStockCandidates is unavailable.");
        }
        object candidates = Activator.CreateInstance(candidateList);
        copy.Invoke(itemRuntime, new[] { Enum.Parse(category, categoryName), candidates });
        Dictionary<string, object> stacks = ((IEnumerable)InvokePublicNoArgument(itemRuntime, "GetAllStacks"))
            .Cast<object>()
            .ToDictionary(value => Convert.ToString(P(value, "StackId")), StringComparer.Ordinal);
        int total = 0;
        var rows = new List<string>();
        foreach (object value in (IEnumerable)candidates)
        {
            if (!(bool)P(value, "IsValid"))
            {
                continue;
            }
            string stackId = Convert.ToString(P(value, "StackId"));
            if (!stacks.TryGetValue(stackId, out object stack))
            {
                continue;
            }
            int available = Convert.ToInt32(P(stack, "AvailableQuantity"));
            if (available <= 0)
            {
                continue;
            }
            total += available;
            rows.Add(stackId + "=" + Convert.ToString(P(stack, "ItemId"))
                + ":" + available);
        }
        detail = rows.Count == 0 ? "<none>" : string.Join("|", rows);
        return total;
    }

    private void SubscribeResearchProgress()
    {
        object bus = Resolve("DungeonStory.Foundation.IGameEventBus", "DungeonStory.Foundation");
        Type eventType = T("ResearchProgressEvent", "Assembly-CSharp");
        var subscribe = bus.GetType().GetMethods().Single(value => value.Name == "Subscribe"
            && value.IsPublic && value.IsGenericMethodDefinition
            && value.GetParameters().Length == 1);
        var callback = GetType().GetMethod("OnResearchProgress");
        if (callback == null || !callback.IsPublic || !callback.IsGenericMethodDefinition)
        {
            throw new InvalidOperationException("Public research event callback is unavailable.");
        }
        Delegate handler = Delegate.CreateDelegate(typeof(Action<>).MakeGenericType(eventType), this,
            callback.MakeGenericMethod(eventType));
        progressSubscription = (IDisposable)subscribe.MakeGenericMethod(eventType)
            .Invoke(bus, new object[] { handler });
        if (progressSubscription == null)
        {
            throw new InvalidOperationException("Research progress subscription was not created.");
        }
    }

    public void OnResearchProgress<TEvent>(TEvent value)
    {
        object gameEvent = value;
        string projectId = Convert.ToString(P(gameEvent, "ProjectId"));
        string actorId = IdValue(P(gameEvent, "Researcher"));
        float approved = Convert.ToSingle(P(gameEvent, "ApprovedWork"));
        float delta = Convert.ToSingle(P(gameEvent, "ProgressDelta"));
        int day = Convert.ToInt32(P(gameEvent, "AbsoluteDay"));
        eventRecords.Add("actor=" + actorId + ";project=" + projectId
            + ";approved=" + approved + ";delta=" + delta + ";day=" + day);
        if (!ProjectIds.Contains(projectId, StringComparer.Ordinal))
        {
            return;
        }
        if (!string.Equals(actorId, selectedActorId, StringComparison.Ordinal))
        {
            unexpectedChainContributor = true;
            return;
        }
        approvedByProject[projectId] = approvedByProject.TryGetValue(projectId, out float prior)
            ? prior + approved
            : approved;
        deltaByProject[projectId] = deltaByProject.TryGetValue(projectId, out float priorDelta)
            ? priorDelta + delta
            : delta;
    }

    private void RequireProjectCapacity(object project, int expectedIndex)
    {
        string id = ProjectIdOf(project);
        float work = Convert.ToSingle(P(project, "RequiredWork"));
        int maximum = Convert.ToInt32(P(project, "MaximumResearchers"));
        object capacity = P(runtime, "ResearchFacilityCapacity");
        Type capability = T("ResearchFacilityCapabilityId", "DungeonStory.Research");
        var available = capacity.GetType().GetMethod("GetAvailable", new[] { capability });
        var meets = capacity.GetType().GetMethods().Single(method => method.Name == "MeetsRequirements"
            && method.IsPublic && method.GetParameters().Length == 2
            && method.GetParameters()[0].ParameterType == T("ResearchProjectSO", "DungeonStory.Research")
            && method.GetParameters()[1].ParameterType == typeof(string).MakeByRefType());
        if (available == null || !available.IsPublic)
        {
            throw new InvalidOperationException("Public research capacity query is unavailable.");
        }
        int basic = Convert.ToInt32(available.Invoke(capacity, new[]
        {
            Enum.Parse(capability, "Basic")
        }));
        object[] requirements = { project, null };
        bool met = (bool)meets.Invoke(capacity, requirements);
        projectContracts.Add(id + ":work=" + work + ";max=" + maximum
            + ";basicAvailable=" + basic + ";requirementsMet=" + met
            + ";blocker=" + Convert.ToString(requirements[1]));
        if (!string.Equals(id, ProjectIds[expectedIndex], StringComparison.Ordinal)
            || !Mathf.Approximately(work, RequiredWork[expectedIndex])
            || maximum != 1 || basic < 1 || !met)
        {
            throw new InvalidOperationException(
                "WIM017053_PREPARATION_BLOCKER_AUTHORED_PROJECT_CAPACITY;"
                + projectContracts.Last());
        }
    }

                + ";expectedActive=" + expectedActiveProjectId);

    private static object F(object value, string name)
    {
        var field = value?.GetType().GetField(name);
        if (field == null || !field.IsPublic)
        {
            throw new InvalidOperationException("Missing public field: " + name);
        }
        return field.GetValue(value);
    }

    private static object PF(object value, string name)
    {
        var property = value?.GetType().GetProperty(name);
        if (property != null && property.GetMethod != null && property.GetMethod.IsPublic)
        {
            return property.GetValue(value);
        }
        var field = value?.GetType().GetField(name);
        if (field != null && field.IsPublic)
        {
            return field.GetValue(value);
        }
        throw new InvalidOperationException("Missing public member: " + name);
    }

    private static object InvokePublic(object value, string name, params object[] arguments)
    {
        var methods = value?.GetType().GetMethods()
            .Where(method => method.IsPublic && !method.IsStatic
                && string.Equals(method.Name, name, StringComparison.Ordinal)
                && method.GetParameters().Length == arguments.Length)
            .ToArray() ?? Array.Empty<System.Reflection.MethodInfo>();
        if (methods.Length != 1)
        {
            throw new InvalidOperationException(
                "Expected one public " + name + " overload; count=" + methods.Length);
        }
        return methods[0].Invoke(value, arguments);
    }

    private static object InvokeStatic(Type type, string name, params object[] arguments)
    {
        var methods = type.GetMethods()
            .Where(method => method.IsPublic && method.IsStatic
                && string.Equals(method.Name, name, StringComparison.Ordinal)
                && method.GetParameters().Length == arguments.Length)
            .ToArray();
        if (methods.Length != 1)
        {
            throw new InvalidOperationException(
                "Expected one public static " + type.FullName + "." + name
                + " overload; count=" + methods.Length);
        }
        return methods[0].Invoke(null, arguments);
    }

    private static object FindProject(object runtime, string id)
    {
        object catalog = P(runtime, "ProjectCatalog");

        var method = catalog.GetType().GetMethods().Single(candidate =>

            && candidate.GetParameters().Length == 2
            && candidate.GetParameters()[0].ParameterType == idType
            && candidate.GetParameters()[1].IsOut);

        {
            throw new InvalidOperationException("Authored research project missing: " + id);

        return arguments[1];
    }

    private void CompleteResearchFixture()
    {
        researchRuntime = FindLoadedComponent("BlueprintResearchRuntime");
        object state = P(researchRuntime, "State");
        var completion = researchRuntime.GetType().GetMethods().Single(method =>
            method.IsPublic && !method.IsStatic
            && string.Equals(method.Name,
                "TryCompleteProjectImmediatelyForVerification",
                StringComparison.Ordinal)
            && method.GetParameters().Length == 2
            && method.GetParameters()[0].ParameterType
                == T("ResearchProjectId", "DungeonStory.Research")
            && method.GetParameters()[1].IsOut);

        for (int index = 0; index < ProjectIds.Length; index++)
        {
            string id = ProjectIds[index];
            object project = FindProject(researchRuntime, id);
            string[] declared = Items(P(project, "Prerequisites"))
                .Select(value => IdValue(P(value, "ProjectId")))
                .ToArray();
            if (!declared.SequenceEqual(ExpectedPrerequisites[index],
                    StringComparer.Ordinal))
            {
                throw new InvalidOperationException(
                    "WIM017053_FIXTURE_DECLARED_PREREQUISITE_MISMATCH;project=" + id
                    + ";expected=" + string.Join(",", ExpectedPrerequisites[index])
                    + ";actual=" + string.Join(",", declared));
            }
            if (Mathf.Abs(Convert.ToSingle(P(project, "RequiredWork"))
                    - ExpectedWork[index]) > FloatTolerance
                || Convert.ToInt32(P(project, "MaximumResearchers")) != 1)
            {
                throw new InvalidOperationException(
                    "WIM017053_FIXTURE_AUTHORED_WORK_OR_CAP_MISMATCH;project=" + id);
            }
            if (ProjectCompleted(researchRuntime, id))
            {
                throw new InvalidOperationException(
                    "WIM017053_FIXTURE_PROJECT_NOT_FRESH;project=" + id);
            }
            foreach (string prerequisite in declared)
            {
                if (!ProjectCompleted(researchRuntime, prerequisite))
                {
                    throw new InvalidOperationException(
                        "WIM017053_FIXTURE_PREREQUISITE_NOT_COMPLETED;project=" + id
                        + ";prerequisite=" + prerequisite);
                }
            }

            int[] authoredBuildings = Items(P(project, "Unlocks"))
                .Where(value => value.GetType().Name == "BlueprintBuildingUnlock")
                .Select(value => Convert.ToInt32(P(value, "BuildingId")))
                .ToArray();
            string[] authoredRecipes = Items(P(project, "Unlocks"))
                .Where(value => value.GetType().Name == "BlueprintRecipeUnlock")
                .Select(value => Convert.ToString(F(value, "recipeId")))
                .ToArray();
            if (!authoredBuildings.SequenceEqual(ExpectedBuildingUnlocks[index])
                || !authoredRecipes.SequenceEqual(
                    ExpectedRecipeUnlocks[index], StringComparer.Ordinal))
            {
                throw new InvalidOperationException(
                    "WIM017053_FIXTURE_AUTHORED_UNLOCK_MISMATCH;project=" + id);
            }

            int blueprintId = Convert.ToInt32(P(project, "BlueprintId"));
            bool expectsBlueprint = string.Equals(
                id, "research:survival:support", StringComparison.Ordinal);
            if ((expectsBlueprint && (blueprintId != 6103
                    || !string.Equals(Convert.ToString(P(project, "BlueprintRule")),
                        "Required", StringComparison.Ordinal)))
                || (!expectsBlueprint && blueprintId != -1))
            {
                throw new InvalidOperationException(
                    "WIM017053_FIXTURE_BLUEPRINT_CONTRACT_MISMATCH;project=" + id
                    + ";blueprint=" + blueprintId);
            }

            object[] arguments = { ProjectId(id), string.Empty };
            bool completed = (bool)completion.Invoke(researchRuntime, arguments);
            string reason = Convert.ToString(arguments[1]);
            if (!completed || !string.IsNullOrEmpty(reason)
                || !ProjectCompleted(researchRuntime, id))
            {
                throw new InvalidOperationException(
                    "WIM017053_FIXTURE_CANONICAL_COMPLETION_FAILED;project=" + id
                    + ";reason=" + reason);
            }

            foreach (int buildingId in ExpectedBuildingUnlocks[index])
            {
                if (!ContainsInt(P(state, "UnlockedBuildingIds"), buildingId))
                {
                    throw new InvalidOperationException(
                        "WIM017053_FIXTURE_BUILDING_UNLOCK_MISSING;project=" + id
                        + ";building=" + buildingId);
                }
            }
            foreach (string recipeId in ExpectedRecipeUnlocks[index])
            {
                if (!ContainsString(P(state, "UnlockedRecipeIds"), recipeId))
                {
                    throw new InvalidOperationException(
                        "WIM017053_FIXTURE_RECIPE_UNLOCK_MISSING;project=" + id
                        + ";recipe=" + recipeId);
                }
            }
            if (expectsBlueprint
                && !ContainsInt(P(state, "CompletedBlueprintIds"), 6103))
            {
                throw new InvalidOperationException(
                    "WIM017053_FIXTURE_BLUEPRINT6103_COMPLETION_MISSING.");
            }
            fixtureRows.Add(id + "{prerequisites=" + string.Join(",", declared)
                + ";work=" + ExpectedWork[index]
                + ";max=1;blueprint=" + blueprintId
                + ";buildings=" + string.Join(",", authoredBuildings)
                + ";recipes=" + string.Join(",", authoredRecipes) + "}");
        }
    }

    private IEnumerator PrepareLossEquipment(object[] actors)
    {

    private void ApplyPostBattleDeathFixture(object expedition)
    {
        if (Items(P(expedition, "EquipmentBaselines")).Count(value =>
                string.Equals(Convert.ToString(P(value, "InstanceId")), lossInstanceId, StringComparison.Ordinal)) != 1)
            throw new InvalidOperationException("WIM053_LOSS_INSTANCE_NOT_IN_LAUNCH_BASELINE.");
        object[] members = Items(P(expedition, "MemberActors")).ToArray();
        object actor = members.SingleOrDefault(value =>
            string.Equals(ActorId(value), lossActorId, StringComparison.Ordinal))
            ?? throw new InvalidOperationException("WIM053_BASELINED_MEMBER_MISSING_AFTER_BATTLE.");
        object[] living = members.Where(value => !(bool)P(value, "IsDead")).ToArray();
        object equipment = Resolve("ICombatEquipmentRuntime");
        object[] currentArgs = { lossInstanceId, null };
        if (!(bool)InvokePublic(equipment, "TryGetInstance", currentArgs))
            throw new InvalidOperationException("WIM053_BASELINED_GEAR_MISSING_AFTER_BATTLE.");
        if ((bool)P(actor, "IsDead"))
        {
            if (living.Length == 0
                || !string.Equals(Convert.ToString(F(currentArgs[1], "worldState")), "Lost", StringComparison.Ordinal))
                throw new InvalidOperationException("WIM053_PREEXISTING_DEATH_WITHOUT_SURVIVOR_OR_EXACT_GEAR_LOSS.");
            deathLossObserved = true;
            Debug.Log("WIM053_ACTUAL_DEATH_GEAR_LOSS;actor=" + lossActorId
                + ";instance=" + lossInstanceId + ";worldState=Lost"
                + ";alreadyDeadAfterBattle=true;additionalDamageApplied=false;naturalDeathCauseNotAsserted=true");
            return;
        }
        if (living.Length < 2) throw new InvalidOperationException("WIM053_NO_SURVIVOR_FOR_DEATH_FIXTURE.");
        if (!string.Equals(Convert.ToString(F(currentArgs[1], "worldState")), "Equipped", StringComparison.Ordinal))
            throw new InvalidOperationException("WIM053_BASELINED_GEAR_NOT_EQUIPPED_BEFORE_DEATH.");
        object anatomy = Resolve("IAnatomyHealthRuntime");
        object snapshot = InvokeActorQuery(anatomy, "GetAnatomySnapshot", actor);
        object[] profileArgs = { Convert.ToString(P(snapshot, "ProfileId")), null };
        if (!(bool)InvokePublic(Resolve("IAnatomyProfileCatalog", "DungeonStory.Medical"), "TryGet", profileArgs))
            throw new InvalidOperationException("WIM053_CURRENT_ANATOMY_PROFILE_MISSING.");
        object profile = profileArgs[1];
        object vital = Items(P(snapshot, "Nodes")).FirstOrDefault(node => {
            object[] args = { Convert.ToString(F(node, "nodeId")), null };
            return !(bool)F(node, "missing") && Convert.ToSingle(F(node, "currentHealth")) > 0f
                && (bool)InvokePublic(profile, "TryGetNode", args) && (bool)P(args[1], "Vital");
        });
        if (vital == null) throw new InvalidOperationException("WIM053_LIVE_VITAL_NODE_MISSING.");
        string nodeId = Convert.ToString(F(vital, "nodeId"));
        Debug.Log("WIM053_BEFORE_VITAL_DAMAGE;actor=" + lossActorId + ";profile=" + P(snapshot, "ProfileId")
            + ";node=" + nodeId + ";health=" + F(vital, "currentHealth")
            + ";lifecycle=" + P(actor, "CurrentLifecycleState") + ";battleOutcome=" + observedBattleOutcome);
        if (!(bool)InvokePublic(anatomy, "TryDamageNode", actor, nodeId,
                Convert.ToSingle(F(vital, "currentHealth")), 0f, "WIM053 protected member death fixture"))
            throw new InvalidOperationException("WIM053_VITAL_DAMAGE_COMMAND_REJECTED.");
        currentArgs = new object[] { lossInstanceId, null };
        if (!(bool)P(actor, "IsDead") || !(bool)InvokePublic(equipment, "TryGetInstance", currentArgs)
            || !string.Equals(Convert.ToString(F(currentArgs[1], "worldState")), "Lost", StringComparison.Ordinal)
            || living.Where(value => !string.Equals(ActorId(value), lossActorId, StringComparison.Ordinal))
                .All(value => (bool)P(value, "IsDead")))
            throw new InvalidOperationException("WIM053_DEATH_EVENT_GEAR_LOSS_NOT_COMMITTED.");
        deathLossObserved = true;
        Debug.Log("WIM053_ACTUAL_DEATH_GEAR_LOSS;actor=" + lossActorId + ";node=" + nodeId
            + ";instance=" + lossInstanceId + ";worldState=Lost;naturalDeathEvidence=false");
    }

    private string ActiveCasualtyFingerprint(object captured)
    {
        object run = FindActiveExpedition(expeditionId)
            ?? throw new InvalidOperationException("WIM053_CASUALTY_ACTIVE_RUN_MISSING");
        object[] members = Items(P(run, "MemberActors")).ToArray();
        object casualty = members.Single(value => ActorId(value) == lossActorId);
        if (!(bool)P(casualty, "IsDead") || members.All(value => (bool)P(value, "IsDead"))
            || Items(P(run, "EquipmentBaselines")).Count(value =>
                Convert.ToString(P(value, "InstanceId")) == lossInstanceId) != 1)
            throw new InvalidOperationException("WIM053_CASUALTY_OR_SURVIVOR_BASELINE_CHANGED");
        object equipment = Resolve("ICombatEquipmentRuntime");
        object[] instanceArgs = { lossInstanceId, null };
        if (!(bool)InvokePublic(equipment, "TryGetInstance", instanceArgs)
            || Convert.ToString(F(instanceArgs[1], "worldState")) != "Lost")
            throw new InvalidOperationException("WIM053_CASUALTY_EXACT_LOST_INSTANCE_CHANGED");
        object equipmentSave = InvokePublic(equipment, "Capture");
        if (completedEnemyIds.Length == 0 || Items(F(equipmentSave, "loadouts")).Any(row =>
                completedEnemyIds.Contains(Convert.ToString(F(row, "characterId")), StringComparer.Ordinal)))
            throw new InvalidOperationException("WIM053_COMPLETED_ENEMY_LOADOUT_NOT_RETIRED");
        string[] recoveredIds = Items(P(run, "RecoveredEquipmentInstanceIds"))
            .Select(Convert.ToString).OrderBy(id => id, StringComparer.Ordinal).ToArray();
        if (recoveredIds.Distinct(StringComparer.Ordinal).Count() != recoveredIds.Length)
            throw new InvalidOperationException("WIM053_ACTUAL_RECOVERED_EQUIPMENT_DUPLICATED");
        object[] physical = Items(InvokePublic(Resolve("IWorldItemStackRuntime"), "GetAllStacks")).ToArray();
        var recoveredRows = new List<string>();
        foreach (string id in recoveredIds)
        {
            object[] pendingArgs = { id, null };
            if (!(bool)InvokePublic(equipment, "TryGetInstance", pendingArgs)
                || Convert.ToString(F(pendingArgs[1], "worldState")) != "ExpeditionPacked"
                || !string.IsNullOrEmpty(Convert.ToString(F(pendingArgs[1], "ownerCharacterId")))
                || !string.IsNullOrEmpty(Convert.ToString(F(pendingArgs[1], "sourceStackId")))
                || physical.Any(stack => Convert.ToString(P(stack, "ItemInstanceId")) == id))
                throw new InvalidOperationException("WIM053_PENDING_RECOVERED_CUSTODY_INVALID:" + id);
            recoveredRows.Add(id + ":" + JsonUtility.ToJson(pendingArgs[1]));
        }
        object section = Items(F(captured, "sections")).Single(row =>
            Convert.ToString(F(row, "sectionId")) == "characters.world");
        object savedWorld = JsonUtility.FromJson(Convert.ToString(F(section, "payloadJson")),
            T("DungeonCharacterWorldSaveData", "Assembly-CSharp"));
        var rows = new List<string>();
        foreach (object member in Items(P(run, "MemberStates")).OrderBy(
                     value => ActorId(P(value, "Actor")), StringComparer.Ordinal))
        {
            object actor = P(member, "Actor");
            string id = ActorId(actor);
            object saved = Items(F(savedWorld, "actors")).Single(row =>
                Convert.ToString(F(row, "persistentId")) == id);
            if (Convert.ToString(F(saved, "lifecycleState")) != Convert.ToString(P(actor, "CurrentLifecycleState")))
                throw new InvalidOperationException("WIM053_CASUALTY_SAVED_LIFECYCLE_MISMATCH:" + id);
            rows.Add(id + ":" + P(actor, "CurrentLifecycleState") + ":" + P(member, "IsAlive")
                + ":" + P(member, "Formation") + ":" + P(member, "Stress") + ":" + P(member, "TotalDamageTaken")
                + ":" + F(saved, "currentHealth") + ":" + F(saved, "injurySeverity")
                + ":" + JsonUtility.ToJson(F(saved, "carryInventory")));
        }
        return string.Join("|", rows) + "|lost=" + JsonUtility.ToJson(instanceArgs[1])
            + "|equipment=" + JsonUtility.ToJson(equipmentSave)
            + "|pendingRecovered=" + string.Join("|", recoveredRows);
    }

    private static string CasualtySectionFingerprint(object captured)
    {
        return string.Join("|", Items(F(captured, "sections"))
            .OrderBy(row => Convert.ToString(F(row, "sectionId")), StringComparer.Ordinal)
            .Select(row => JsonUtility.ToJson(row)));
    }

    private void VerifyLivingTombstoneRejected(object saves, object captured, string validJson)
    {
        string before = CasualtySectionFingerprint(captured);
        int events = rewardEventIds.Count;
        object invalid = InvokePublic(saves, "FromJson", validJson);
        object envelope = Items(F(invalid, "sections")).Single(row =>
            Convert.ToString(F(row, "sectionId")) == "characters.world");
        object worldCopy = JsonUtility.FromJson(Convert.ToString(F(envelope, "payloadJson")),
            T("DungeonCharacterWorldSaveData", "Assembly-CSharp"));
        object casualtyCopy = Items(F(worldCopy, "actors")).Single(row =>
            Convert.ToString(F(row, "persistentId")) == lossActorId);
        if (Convert.ToString(F(casualtyCopy, "lifecycleState")) != "Despawned"
            || Convert.ToSingle(F(casualtyCopy, "currentHealth")) > 0f)
            throw new InvalidOperationException("WIM053_NEGATIVE_REQUIRES_ACTUAL_SAVED_CASUALTY");
        var lifecycleField = casualtyCopy.GetType().GetField("lifecycleState");
        lifecycleField.SetValue(casualtyCopy, Enum.Parse(lifecycleField.FieldType, "Active", false));
        casualtyCopy.GetType().GetField("currentHealth").SetValue(casualtyCopy, 1f);
        envelope.GetType().GetField("payloadJson").SetValue(envelope, JsonUtility.ToJson(worldCopy));
        object[] args = { invalid, null };
        bool accepted;
        try { accepted = (bool)InvokePublic(saves, "TryRestore", args); }
        finally { Time.timeScale = 0f; }
        string expected = "Character '" + lossActorId + "' is both living and archived as a tombstone.";
        if (accepted || args[1] == null || (bool)P(args[1], "Success")
            || !Items(P(args[1], "Errors")).Any(error =>
                string.Equals(Convert.ToString(error), expected, StringComparison.Ordinal)))
            throw new InvalidOperationException("WIM053_LIVING_TOMBSTONE_ACCEPTED_OR_WRONG_REJECTION;"
                + (args[1] == null ? "no-report" : string.Join("|", Items(P(args[1], "Errors")).Select(Convert.ToString))));
        // No yield: a rejected restore must not publish any section or reward change.
        // Ordinary wildlife RNG may advance on a player frame even at timeScale zero.
        object unchanged = InvokePublic(saves, "Capture");
        if (!string.Equals(before, CasualtySectionFingerprint(unchanged), StringComparison.Ordinal)
            || rewardEventIds.Count != events)
            throw new InvalidOperationException("WIM053_LIVING_TOMBSTONE_REJECT_CHANGED_LIVE_STATE");
        Debug.Log("WIM053_LIVING_TOMBSTONE_REJECTED_ATOMIC;actor=" + lossActorId
            + ";allSectionsExact=true;rewardEventsUnchanged=true;scope=synchronous-public-restore");
    }

    private IEnumerator VerifyActiveCasualtyRoundtrip()
    {
        float previousScale = Time.timeScale;
        Time.timeScale = 0f;
        try
        {
            yield return null; // Complete real battle cleanup before capturing its owners.
            ResolveJourneyAuthorities();
            if ((bool)P(battle, "HasActiveBattle"))
                throw new InvalidOperationException("WIM053_CASUALTY_BATTLE_NOT_COMPLETE");
            object saves = Resolve("IDungeonGameSaveService", "DungeonStory.Infrastructure");
            object captured = InvokePublic(saves, "Capture");
            string before = ActiveCasualtyFingerprint(captured);
            string offenseBefore = CaptureOffenseJson();
            int events = rewardEventIds.Count;
            string json = Convert.ToString(InvokePublic(saves, "ToJson", captured, false));
            VerifyLivingTombstoneRejected(saves, captured, json);
            object[] args = { InvokePublic(saves, "FromJson", json), null };
            bool restored;
            try { restored = (bool)InvokePublic(saves, "TryRestore", args); }
            finally { Time.timeScale = 0f; }
            if (!restored || args[1] == null || !(bool)P(args[1], "Success"))
                throw new InvalidOperationException("WIM053_ACTIVE_CASUALTY_FULLSAVE_FAILED;"
                    + (args[1] == null ? "no-report" : string.Join("|", Items(P(args[1], "Errors")).Select(Convert.ToString))));
            yield return null;
            ResolveJourneyAuthorities();
            object recaptured = InvokePublic(Resolve("IDungeonGameSaveService", "DungeonStory.Infrastructure"), "Capture");
            if (!string.Equals(before, ActiveCasualtyFingerprint(recaptured), StringComparison.Ordinal)
                || !string.Equals(offenseBefore, CaptureOffenseJson(), StringComparison.Ordinal)
                || rewardEventIds.Count != events)
                throw new InvalidOperationException("WIM053_ACTIVE_CASUALTY_RESTORE_CHANGED_STATE_OR_REGRANTED");
            casualtyRoundtripObserved = true;
            Debug.Log("WIM053_ACTIVE_CASUALTY_FULLSAVE_PASS;expedition=" + expeditionId
                + ";actor=" + lossActorId + ";lostInstance=" + lossInstanceId
                + ";memberStatesExact=true;survivorPreserved=true;enemyLoadouts=0;offenseExact=true;rewardEventsUnchanged=true");
            int recoveredCount = CountOf(P(FindActiveExpedition(expeditionId), "RecoveredEquipmentInstanceIds"));
            Debug.Log((recoveredCount > 0 ? "WIM053_PENDING_RECOVERED_FULLSAVE_PASS" : "WIM053_PENDING_RECOVERED_NOT_OBSERVED")
                + ";expedition=" + expeditionId + ";count=" + recoveredCount
                + ";scope=actual-present-recovered-items-only;zeroIsNotPositiveRecoveryEvidence=true");
        }
        finally { Time.timeScale = previousScale; }
    }

    private IEnumerator ObserveUncontainedArrival()
    {
        float until = Time.realtimeSinceStartup + 170f;
        Time.timeScale = 1f;
        bool physicalObserved = false;
        float nextPresenceCheck = 0f;
        while (Time.realtimeSinceStartup < until)
        {
            object[] rows = Items(P(returnArrivals, "Arrivals")).Where(row =>
                string.Equals(Convert.ToString(F(row, "expeditionId")), expeditionId, StringComparison.Ordinal)).ToArray();
            foreach (object row in rows)
            {
                var ids = Items(F(row, "materializedIds")).Select(Convert.ToString).ToArray();
                if (!physicalObserved && ids.Length > 0)
                {
                    var live = Resources.FindObjectsOfTypeAll(T("CharacterActor", "Assembly-CSharp"))
                        .OfType<Component>().Where(actor => actor.gameObject.scene.isLoaded).ToArray();
                    if (ids.All(id => live.Any(actor => string.Equals(ActorId(actor), id, StringComparison.Ordinal)
                            && !(bool)P(actor, "IsDead"))))
                    {
                        physicalObserved = true;
                        Debug.Log("WIM053_REAL_PRISONER_ARRIVAL;expedition=" + expeditionId
                            + ";arrival=" + F(row, "arrivalId") + ";actors=" + string.Join(",", ids));
                    }
                }
            }
            var receipts = Items(InvokePublic(returnArrivals, "GetSettlementReceipts", expeditionId));
            if (receipts.Count > 0 && receipts.All(value =>
                    !string.Equals(Convert.ToString(P(value, "resolution")), "Pending", StringComparison.Ordinal)))
            {
                if (!physicalObserved || receipts.Any(value =>
                        Convert.ToInt32(P(value, "materializedAmount")) <= 0
                        || Convert.ToInt32(P(value, "escapedAmount")) != Convert.ToInt32(P(value, "materializedAmount"))
                        || Convert.ToInt32(P(value, "securedAmount")) != 0))
                    throw new InvalidOperationException("WIM053_EXPECTED_UNCONTAINED_ESCAPE_NOT_OBSERVED.");
                escapedArrivalFingerprint = ArrivalFingerprint(receipts);
                escapeObserved = true;
                Debug.Log("WIM053_ACTUAL_ARRIVAL_ESCAPE;" + escapedArrivalFingerprint);
                Time.timeScale = 0f;
                yield break;
            }
            // Check after terminal receipts: legitimate escape destroys the actor
            // and publishes its receipt synchronously. A missing pending actor is
            // already a failed fixture and must not consume the entire deadline.
            if (physicalObserved && Time.realtimeSinceStartup >= nextPresenceCheck)
            {
                nextPresenceCheck = Time.realtimeSinceStartup + 0.5f;
                var live = Resources.FindObjectsOfTypeAll(T("CharacterActor", "Assembly-CSharp"))
                    .OfType<Component>().Where(actor => actor.gameObject.scene.isLoaded).ToArray();
                foreach (object row in rows)
                {
                    var escaped = Items(F(row, "escapedIds")).Select(Convert.ToString).ToArray();
                    foreach (string id in Items(F(row, "materializedIds")).Select(Convert.ToString)
                                 .Where(id => !escaped.Contains(id, StringComparer.Ordinal)))
                    {
                        Component actor = live.SingleOrDefault(value =>
                            string.Equals(ActorId(value), id, StringComparison.Ordinal));
                        if (actor == null || (bool)P(actor, "IsDead"))
                            throw new InvalidOperationException("WIM053_PENDING_ARRIVAL_SUBJECT_LOST;actor=" + id
                                + ";missing=" + (actor == null) + ";unityTime=" + Time.time
                                + ";risk=" + F(row, "escapeRisk") + ";status=" + F(row, "lastStatus"));
                    }
                }
            }
            yield return null;
        }
        Time.timeScale = 0f;
        throw new InvalidOperationException("WIM053_ARRIVAL_ESCAPE_DEADLINE;");
    }

    private object SelectTarget(object[] availableActors, out object site)
    {
        foreach (object candidateSite in Items(P(world, "Sites"))
                     .Where(value => (bool)P(value, "IsActive")
                         && (!LossArrivalScenario || string.Equals(Convert.ToString(F(value, "archetypeId")),
                             "prisoner_convoy", StringComparison.Ordinal))
                         && string.Equals(Convert.ToString(F(value, "state")),
                             "Revealed", StringComparison.Ordinal))
                     .OrderBy(value => Convert.ToSingle(F(value, "strength")))
                     .ThenBy(value => Convert.ToString(F(value, "siteId")),
                         StringComparer.Ordinal))
        {
            object[] arguments =
            {
                Convert.ToString(F(candidateSite, "siteId")),
                null,
                Activator.CreateInstance(
                    T("OffenseHexCoord", "DungeonStory.Foundation"))
            };
            bool created = (bool)InvokePublic(targets, "TryCreateTarget", arguments);
            object candidateTarget = arguments[1];
            if (created && candidateTarget != null
                && availableActors.Length >= Convert.ToInt32(F(candidateTarget, "requiredMembers")))
            {
                site = candidateSite;
                return candidateTarget;
            }
        }
        site = null;
        return null;
    }

    private static object InvokeActorQuery(object query, string methodName, object actor)
    {
        Type actorType = T("CharacterActor", "Assembly-CSharp");
        var method = query.GetType().GetMethods().SingleOrDefault(value =>
            value.IsPublic && !value.IsStatic
            && string.Equals(value.Name, methodName, StringComparison.Ordinal)
            && value.GetParameters().Length == 1
            && value.GetParameters()[0].ParameterType == actorType);
        if (method == null || !actorType.IsInstanceOfType(actor))
            throw new InvalidOperationException("WIM053_ACTOR_QUERY_SIGNATURE_MISMATCH:" + methodName);
        return method.Invoke(query, new[] { actor });
    }

                string.Equals(ActorId(value), fieldActorId, StringComparison.Ordinal));
            object anatomy = Resolve("IAnatomyHealthRuntime");
            object body = Resolve("ICharacterBodyHealthQuery");
            object medical = Resolve("IOffenseFieldMedicalRuntime");
            object before = InvokeActorQuery(anatomy, "GetAnatomySnapshot", member);
            object node = Items(P(before, "Nodes")).Single(value =>
                string.Equals(Convert.ToString(F(value, "nodeId")), FieldNodeId,
                    StringComparison.Ordinal));
            float damage = Convert.ToSingle(F(node, "currentHealth"));
            if ((bool)P(member, "IsDead") || (bool)P(InvokeActorQuery(body, "GetSnapshot", member), "Downed")
                || (bool)F(node, "missing") || damage <= 0f
                || !string.Equals(Convert.ToString(P(before, "ProfileId")),
                    "anatomy:slime", StringComparison.Ordinal))
                throw new InvalidOperationException("WIM053_FIELD_DAMAGE_PREFLIGHT_FAILED.");
            int stockBefore = Convert.ToInt32(InvokePublic(P(expedition, "Supplies"), "Get", fieldKitType));
            CaptureExpectedConsumption(expedition);
            expectedConsumption.TryGetValue(fieldKitItemId, out int consumedBefore);
            int treatmentsBefore = CountOf(P(expedition, "TreatmentReceipts"));
            if (stockBefore < 1) throw new InvalidOperationException("WIM053_PACKED_FIELD_KIT_MISSING.");
            if (!(bool)InvokePublic(anatomy, "TryDamageNode", member, FieldNodeId,
                    damage, 0f, "WIM053 protected nonvital field treatment fixture"))
                throw new InvalidOperationException("WIM053_FIELD_DAMAGE_COMMAND_FAILED.");
            object damaged = InvokeActorQuery(anatomy, "GetAnatomySnapshot", member);
            object damagedNode = Items(P(damaged, "Nodes")).Single(value =>
                string.Equals(Convert.ToString(F(value, "nodeId")), FieldNodeId,
                    StringComparison.Ordinal));
            if ((bool)P(member, "IsDead") || (bool)P(InvokeActorQuery(body, "GetSnapshot", member), "Downed")
                || (bool)F(damagedNode, "missing")
                || Convert.ToSingle(F(damagedNode, "currentHealth")) > 0.01f)
                throw new InvalidOperationException("WIM053_FIELD_DAMAGE_UNSAFE_OR_NOT_APPLIED.");
            InvokePublic(panels, "ShowWorldMap");
            yield return null;
            Canvas.ForceUpdateCanvases();
            string kitLabel = Convert.ToString(InvokeStatic(
                T("OffenseSupplyCatalog", "Assembly-CSharp"), "GetDisplayName", fieldKitType));
            object progression = P(member, "Progression");
            int level = progression == null ? 1 : Convert.ToInt32(P(progression, "Level"));
            object identity = P(member, "Identity");
            string actorName = identity == null ? ((UnityEngine.Object)member).name
                : Convert.ToString(P(identity, "DisplayName"));
            string actorLabel = string.Format(System.Globalization.CultureInfo.CurrentCulture,
                "Lv.{0} {1} · 체력 {2:0}", level, actorName, P(member, "CurrentHealth"));
            string label = "가고정 · " + actorLabel + " · 위족 [" + kitLabel + "]";
            ClickExact(label, "actual-field-stabilization");
            yield return null;
            CaptureExpectedConsumption(expedition);
            expectedConsumption.TryGetValue(fieldKitItemId, out int consumedAfter);
            int stockAfter = Convert.ToInt32(InvokePublic(P(expedition, "Supplies"), "Get", fieldKitType));
            object stabilization = Items(InvokePublic(medical, "GetStabilizations", expeditionId))
                .Single(value => string.Equals(Convert.ToString(F(value, "characterId")), fieldActorId,
                        StringComparison.Ordinal)
                    && string.Equals(Convert.ToString(F(value, "anatomyNodeId")), FieldNodeId,
                        StringComparison.Ordinal));
            if (stockAfter != stockBefore - 1 || consumedAfter != consumedBefore + 1
                || CountOf(P(expedition, "TreatmentReceipts")) != treatmentsBefore + 1
                || !(bool)F(stabilization, "active") || !(bool)F(stabilization, "usedForNode")
                || !Close(Convert.ToSingle(F(stabilization, "locomotionFloor")), 0.5f)
                || !Close(Convert.ToSingle(F(stabilization, "sustainFloor")), 0.5f)
                || string.IsNullOrWhiteSpace(Convert.ToString(F(stabilization, "consumedKitInstanceId"))))
                throw new InvalidOperationException("WIM053_ACTUAL_FIELD_TREATMENT_OR_CONSUMPTION_MISMATCH.");
            Canvas.ForceUpdateCanvases();
            if (ActiveOffenseButtons().Any(button =>
                    string.Equals(Label(button), label, StringComparison.Ordinal)))
                throw new InvalidOperationException("WIM053_USED_FIELD_NODE_STILL_OFFERED.");
            fieldFixtureConsumed = 1;
            fieldMedicalObserved = true;
            fieldMedicalEvidence = "actor=" + fieldActorId + ";node=" + FieldNodeId
                + ";damage=" + damage + ";item=" + fieldKitItemId
                + ";packed=" + stockBefore + "->" + stockAfter
                + ";consumed=" + consumedBefore + "->" + consumedAfter
                + ";treatments=" + treatmentsBefore + "->" + CountOf(P(expedition, "TreatmentReceipts"))
                + ";active=true;used=true;locomotionFloor=0.5;sustainFloor=0.5;duplicateButton=false"
                + ";naturalInjuryEvidence=false";
            Debug.Log("WIM053_FIELD_STABILIZATION_PASS;" + fieldMedicalEvidence);
        }
        finally { Time.timeScale = previousScale; }
    }

    private static string PathFingerprint(object state)
    {
        return string.Join(">", Items(F(state, "remainingPath"))
            .Select(value => Convert.ToInt32(F(value, "q"))
                + "," + Convert.ToInt32(F(value, "r"))));
    }

    private static bool Close(float left, float right)
    {
        return Mathf.Abs(left - right) <= FloatTolerance;
    }

    private void SaveSegment(object state)
    {
        savedCurrentQ = Convert.ToInt32(F(state, "currentQ"));
        savedCurrentR = Convert.ToInt32(F(state, "currentR"));
        savedDestinationQ = Convert.ToInt32(F(state, "destinationQ"));
        savedDestinationR = Convert.ToInt32(F(state, "destinationR"));
        savedDestinationSite = Convert.ToString(F(state, "destinationSiteId"));
        savedPath = PathFingerprint(state);
        savedRoad = Convert.ToSingle(F(state, "routeRoadMultiplier"));
        savedWeather = Convert.ToSingle(F(state, "routeWeatherMultiplier"));
        savedLoad = Convert.ToSingle(F(state, "routeLoadMultiplier"));
        savedSegmentQ = Convert.ToInt32(F(state, "activeSegmentQ"));
        savedSegmentR = Convert.ToInt32(F(state, "activeSegmentR"));
        savedTraversalCost = Convert.ToSingle(F(state, "activeSegmentTraversalCost"));
        savedElapsed = Convert.ToSingle(F(state, "progressToNextTile"));
        savedDuration = Convert.ToSingle(F(state, "activeSegmentDurationSeconds"));
        savedMovement = Convert.ToSingle(F(state, "movementTimeMultiplier"));
        savedMilestone = Convert.ToSingle(F(state, "milestoneTimeMultiplier"));
        savedFacility = Convert.ToSingle(F(state, "facilityTimeMultiplier"));
        savedEventSequence = Convert.ToInt32(F(state, "eventSequence"));
        saveClock = Convert.ToSingle(P(gameClock, "Time"));
    }

    private void RequireSegmentEqual(object state)
    {
        bool exact = Convert.ToInt32(F(state, "currentQ")) == savedCurrentQ
            && Convert.ToInt32(F(state, "currentR")) == savedCurrentR
            && Convert.ToInt32(F(state, "destinationQ")) == savedDestinationQ
            && Convert.ToInt32(F(state, "destinationR")) == savedDestinationR
            && string.Equals(Convert.ToString(F(state, "destinationSiteId")),
                savedDestinationSite, StringComparison.Ordinal)
            && string.Equals(PathFingerprint(state), savedPath, StringComparison.Ordinal)
            && Convert.ToInt32(F(state, "activeSegmentQ")) == savedSegmentQ
            && Convert.ToInt32(F(state, "activeSegmentR")) == savedSegmentR
            && Convert.ToInt32(F(state, "eventSequence")) == savedEventSequence
            && Close(Convert.ToSingle(F(state, "activeSegmentTraversalCost")), savedTraversalCost)
            && Close(Convert.ToSingle(F(state, "progressToNextTile")), savedElapsed)
            && Close(Convert.ToSingle(F(state, "activeSegmentDurationSeconds")), savedDuration)
            && Close(Convert.ToSingle(F(state, "routeRoadMultiplier")), savedRoad)
            && Close(Convert.ToSingle(F(state, "routeWeatherMultiplier")), savedWeather)
            && Close(Convert.ToSingle(F(state, "routeLoadMultiplier")), savedLoad)
            && Close(Convert.ToSingle(F(state, "movementTimeMultiplier")), savedMovement)
            && Close(Convert.ToSingle(F(state, "milestoneTimeMultiplier")), savedMilestone)
            && Close(Convert.ToSingle(F(state, "facilityTimeMultiplier")), savedFacility)
            && !(bool)F(state, "pausedForDecision")
            && !(bool)F(state, "pausedForBattle");
        if (!exact)
        {
            throw new InvalidOperationException(
                "WIM017_SEGMENT_FULLSAVE_RESTORE_MISMATCH;before=" + SegmentSummaryBefore()
                + ";after=" + SegmentSummary(state));
        }
    }

    private void CaptureExpectedConsumption(object expedition)
    {
        expectedConsumption.Clear();
        foreach (object pair in Items(P(expedition, "ConsumedSupplies")))
        {
            object supplyType = P(pair, "Key");
            int quantity = Convert.ToInt32(P(pair, "Value"));
            string itemId = Convert.ToString(InvokeStatic(
                T("OffenseSupplyCatalog", "Assembly-CSharp"),
                "GetPhysicalItemId", supplyType));
            if (quantity > 0)
            {
                expectedConsumption[itemId] = quantity;
            }
        }
    }

    private IEnumerator RestoreCurrentSegment()
    {
        Time.timeScale = 0f;
        object saveService = Resolve(
            "IDungeonGameSaveService", "DungeonStory.Infrastructure");
        object captured = InvokePublic(saveService, "Capture");
        string json = Convert.ToString(InvokePublic(saveService, "ToJson", captured, false));
        object parsed = InvokePublic(saveService, "FromJson", json);
        object[] restoreArguments = { parsed, null };
        bool restored = false;
        try
        {
            restored = (bool)InvokePublic(saveService, "TryRestore", restoreArguments);
        }
        finally
        {
            Time.timeScale = 0f;
        }
        object report = restoreArguments[1];
        if (!restored || report == null || !(bool)P(report, "Success"))
        {
            string errors = report == null
                ? "<no-report>"
                : string.Join(" | ", Items(P(report, "Errors"))
                    .Select(Convert.ToString));
            throw new InvalidOperationException(
                "WIM017_FULL_GAME_SAVE_RESTORE_FAILED;errors=" + errors);
        }
        saveBytes = json.Length;
        yield return null;

        ResolveJourneyAuthorities();
        researchRuntime = FindLoadedComponent("BlueprintResearchRuntime");
        object expedition = FindActiveExpedition(expeditionId);
        if (expedition == null)
        {
            throw new InvalidOperationException(
                "WIM017_FULLSAVE_ACTIVE_EXPEDITION_ID_NOT_REACQUIRED;id=" + expeditionId);
        }
        if (!TryTravelState(out object restoredState))
        {
            throw new InvalidOperationException(
                "WIM017_FULLSAVE_TRAVEL_STATE_NOT_REACQUIRED;id=" + expeditionId);
        }
        RequireSegmentEqual(restoredState);
        for (int index = 0; index < ProjectIds.Length; index++)
        {
            if (!ProjectCompleted(researchRuntime, ProjectIds[index]))
            {
                throw new InvalidOperationException(
                    "WIM017_FULLSAVE_RESEARCH_FIXTURE_MISSING;project=" + ProjectIds[index]);
            }
        }
        object researchState = P(researchRuntime, "State");
        if (!ContainsInt(P(researchState, "CompletedBlueprintIds"), 6103))
        {
            throw new InvalidOperationException(
                "WIM017_FULLSAVE_BLUEPRINT6103_MISSING.");
        }
        segmentRestored = true;
        Debug.Log("WIM017_SEGMENT_FULLSAVE_RESTORE_PASS;before="
            + SegmentSummaryBefore()
            + ";after=" + SegmentSummary(restoredState)
            + ";expeditionId=" + expeditionId + "->"
            + Convert.ToString(P(expedition, "ExpeditionId"))
            + ";sameExpeditionId=" + string.Equals(expeditionId,
                Convert.ToString(P(expedition, "ExpeditionId")),
                StringComparison.Ordinal)
            + ";saveClock=" + saveClock
            + ";jsonChars=" + json.Length);

        float progressDeadline = Time.realtimeSinceStartup + 12f;
        Time.timeScale = 1f;
        while (Time.realtimeSinceStartup < progressDeadline)
        {
            yield return null;
            if (Convert.ToSingle(P(gameClock, "Time")) <= saveClock + FloatTolerance
                || !TryTravelState(out object nextState))
            {
                continue;
            }
            bool elapsedAdvanced = Convert.ToSingle(F(nextState, "progressToNextTile"))
                > savedElapsed + FloatTolerance;
            bool segmentTransitioned = Convert.ToInt32(F(nextState, "currentQ")) != savedCurrentQ
                || Convert.ToInt32(F(nextState, "currentR")) != savedCurrentR
                || Convert.ToInt32(F(nextState, "activeSegmentQ")) != savedSegmentQ
                || Convert.ToInt32(F(nextState, "activeSegmentR")) != savedSegmentR
                || !string.Equals(PathFingerprint(nextState), savedPath, StringComparison.Ordinal);
            if (elapsedAdvanced || segmentTransitioned)
            {
                segmentAdvanced = true;
                break;
            }
        }
        if (!segmentAdvanced)
        {
            throw new InvalidOperationException(
                "WIM017_FULLSAVE_SEGMENT_DID_NOT_RESUME_NORMAL_PROGRESS;clock="
                + saveClock + "->" + P(gameClock, "Time"));
        }
        Debug.Log("WIM017_SEGMENT_NORMAL_PROGRESS_PASS;expeditionId="
            + expeditionId + ";clock=" + saveClock + "->"
            + P(gameClock, "Time") + ";currentSegment="
            + (TryTravelState(out object currentState)
                ? SegmentSummary(currentState)
                : "<missing>"));
    }

    private static float TeamHealth(object session, string team)
    {
        return Items(P(session, "Combatants"))
            .Where(value => string.Equals(Convert.ToString(P(value, "Team")),
                    team, StringComparison.Ordinal)
                && !(bool)P(value, "IsDead"))
            .Sum(value => Mathf.Max(0f, Convert.ToSingle(P(value, "CurrentHealth"))));
    }

    private static int CardRank(object card, object combatant)
    {
        string action = Convert.ToString(F(card, "actionType"));
        if (string.Equals(action, "Advance", StringComparison.Ordinal))
        {
            return string.Equals(Convert.ToString(P(combatant, "Formation")),
                "Front", StringComparison.Ordinal) ? 4 : 0;
        }
        if (string.Equals(action, "BasicAttack", StringComparison.Ordinal)) return 1;
        if (!string.Equals(action, "Ability", StringComparison.Ordinal)) return 4;
        string sourceSkillId = Convert.ToString(F(card, "sourceSkillId"));
        object ability = Items(P(combatant, "Abilities")).FirstOrDefault(value =>
            string.Equals(Convert.ToString(P(value, "Id")), sourceSkillId,
                StringComparison.Ordinal));
        bool damaging = ability != null
            && string.Equals(Convert.ToString(P(ability, "TargetRule")),
                "Enemy", StringComparison.Ordinal)
            && Items(P(ability, "Effects")).Any(effect =>
                effect.GetType().Name == "OffenseDamageEffect"
                && (Convert.ToSingle(P(effect, "BasicDamageMultiplier")) > 0f
                    || Convert.ToSingle(P(effect, "FlatDamage")) > 0f));
        return damaging ? 2 : 3;
    }

    private static int TacticalRank(object card)
    {
        return Convert.ToString(F(card, "tacticalTag")) switch
        {
            "Execute" => 0,
            "Break" => 1,
            "Maneuver" => 2,
            "Intercept" => 3,
            "Support" => 4,
            _ => 5
        };
    }

    private static string TacticalLabel(object value)
    {
        return Convert.ToString(value) switch
        {
            "Intercept" => "저지",
            "Maneuver" => "기동",
            "Break" => "파쇄",
            "Support" => "지원",
            "Execute" => "집행",
            _ => "일반"
        };
    }

    private static void ClickBattleCard(string text, int row)
    {
        TMP_Text deckName = Resources.FindObjectsOfTypeAll<TMP_Text>()
            .SingleOrDefault(value => value != null
                && value.gameObject.scene.isLoaded
                && value.gameObject.activeInHierarchy
                && string.Equals(value.name, "DeckName_" + row,
                    StringComparison.Ordinal));
        if (!(deckName?.transform is RectTransform deckRect))
        {
            throw new InvalidOperationException(
                "WIM053_BATTLE_DECK_ROW_MISSING;row=" + row);
        }
        Button button = ActiveOffenseButtons()
            .Where(value => string.Equals(Label(value), text, StringComparison.Ordinal)
                && value.transform is RectTransform)
            .OrderBy(value => Mathf.Abs(
                ((RectTransform)value.transform).anchorMax.y - deckRect.anchorMax.y))
            .FirstOrDefault();
        if (button == null
            || Mathf.Abs(((RectTransform)button.transform).anchorMax.y
                - deckRect.anchorMax.y) > 0.001f)
        {
            throw new InvalidOperationException(
                "WIM053_BATTLE_CARD_UI_MISSING;row=" + row + ";label=" + text);
        }
        PointerClick(button, "battle-card-row-" + row);
    }

    private IEnumerator IssueBattleTurn()
    {
        object state = P(director, "State");
        object session = P(battle, "Session");
        if (state == null || session == null)
        {
            throw new InvalidOperationException("WIM053_BATTLE_AUTHORITY_MISSING.");
        }
        int turn = Convert.ToInt32(F(state, "turn"));
        int roundBefore = Convert.ToInt32(P(session, "RoundNumber"));
        long commandBefore = Convert.ToInt64(P(session, "LastProcessedCommandId"));
        float allyBefore = TeamHealth(session, "Allies");
        float enemyBefore = TeamHealth(session, "Enemies");
        List<object> decks = Items(F(state, "decks"));
        string deckReadiness = string.Join(",", decks.Select(deck =>
        {
            string id = Convert.ToString(F(deck, "characterId"));
            object actor = FindCombatant(session, id);
            return id + ":candidates=" + Items(F(deck, "candidates")).Count
                + ":canTakeTurn=" + (actor == null ? "missing"
                    : Convert.ToString(P(actor, "CanTakeTurn")));
        }));
        string[] availableDeckIds = decks
            .Where(deck => Items(F(deck, "candidates")).Count > 0
                && (bool)P(FindCombatant(session,
                    Convert.ToString(F(deck, "characterId"))), "CanTakeTurn"))
            .Select(deck => Convert.ToString(F(deck, "characterId")))
            .OrderBy(value => value, StringComparer.Ordinal)
            .ToArray();
        if (availableDeckIds.Length == 0)
        {
            throw new InvalidOperationException("WIM053_BATTLE_NO_AVAILABLE_COMMAND_DECK.");
        }

        List<string> cards = Items(F(state, "commandQueue"))
            .Select(entry => "preserved:"
                + Convert.ToString(F(entry, "characterId")) + ":"
                + Convert.ToString(F(entry, "cardInstanceId")) + "->"
                + Convert.ToString(F(entry, "targetIntentId")))
            .ToList();
        foreach (string actorId in availableDeckIds)
        {
            state = P(director, "State");
            if (state == null || !(bool)P(battle, "HasActiveBattle"))
            {
                throw new InvalidOperationException(
                    "WIM053_BATTLE_ENDED_DURING_COMMAND_SELECTION.");
            }
            if (Items(F(state, "commandQueue")).Any(entry => string.Equals(
                    Convert.ToString(F(entry, "characterId")), actorId,
                    StringComparison.Ordinal)))
            {
                continue;
            }
            decks = Items(F(state, "decks"));
            object deck = decks.Single(value => string.Equals(
                Convert.ToString(F(value, "characterId")), actorId,
                StringComparison.Ordinal));
            int row = decks.IndexOf(deck);
            object combatant = FindCombatant(session, actorId);
            object card = Items(F(deck, "candidates"))
                .OrderBy(value => CardRank(value, combatant))
                .ThenBy(TacticalRank)
                .ThenByDescending(value => Convert.ToSingle(F(value, "power")))
                .ThenByDescending(value => Convert.ToInt32(F(value, "executionStages")))
                .ThenByDescending(value => Convert.ToInt32(F(value, "speed")))
                .ThenBy(value => Convert.ToString(F(value, "instanceId")),
                    StringComparer.Ordinal)
                .FirstOrDefault();
            object intent = Items(F(state, "enemyIntents"))
                .Where(value =>
                {
                    object enemy = FindCombatant(session,
                        Convert.ToString(F(value, "enemyId")));
                    return enemy != null && !(bool)P(enemy, "IsDead")
                        && !(bool)P(enemy, "IsDowned");
                })
                .OrderByDescending(value => Convert.ToSingle(F(value, "threat")))
                .ThenBy(value => Convert.ToString(F(value, "intentId")),
                    StringComparer.Ordinal)
                .FirstOrDefault();
            if (card == null || intent == null)
            {
                if (Items(F(state, "commandQueue")).Count > 0)
                {
                    break;
                }
                throw new InvalidOperationException(
                    "WIM053_BATTLE_NO_INPUTTABLE_COMMAND;actor=" + actorId);
            }
            string cardLabel = Convert.ToString(F(card, "displayName")) + "\n"
                + TacticalLabel(F(card, "tacticalTag")) + " · "
                + F(card, "executionStages") + "단계 · 속도 " + F(card, "speed");
            ClickBattleCard(cardLabel, row);
            yield return null;

            state = P(director, "State");
            string intentId = Convert.ToString(F(intent, "intentId"));
            intent = Items(F(state, "enemyIntents")).FirstOrDefault(value =>
                string.Equals(Convert.ToString(F(value, "intentId")), intentId,
                    StringComparison.Ordinal));
            if (intent == null)
            {
                throw new InvalidOperationException(
                    "WIM053_BATTLE_INTENT_DISAPPEARED_BEFORE_POINTER_COMMIT;id=" + intentId);
            }
            object enemyActor = FindCombatant(session, Convert.ToString(F(intent, "enemyId")));
            string intentLabel = Convert.ToString(P(enemyActor, "DisplayName")) + "\n"
                + TacticalLabel(F(intent, "tacticalTag")) + " "
                + F(intent, "executionStages") + "단계";
            ClickExact(intentLabel, "battle-intent-" + intentId);
            yield return null;

            state = P(director, "State");
            bool queued = Items(F(state, "commandQueue")).Any(entry =>
                string.Equals(Convert.ToString(F(entry, "characterId")), actorId,
                    StringComparison.Ordinal)
                && string.Equals(Convert.ToString(F(entry, "cardInstanceId")),
                    Convert.ToString(F(card, "instanceId")), StringComparison.Ordinal)
                && string.Equals(Convert.ToString(F(entry, "targetIntentId")),
                    intentId, StringComparison.Ordinal));
            if (!queued)
            {
                throw new InvalidOperationException(
                    "WIM053_BATTLE_POINTER_DID_NOT_QUEUE_CARD;actor=" + actorId);
            }
            cards.Add(actorId + ":" + F(card, "displayName") + "->" + intentId);
        }

        state = P(director, "State");
        int queuedCount = Items(F(state, "commandQueue")).Count;
        int deckCount = Items(F(state, "decks")).Count;
        if (queuedCount <= 0)
        {
            throw new InvalidOperationException(
                "WIM053_BATTLE_EXECUTABLE_QUEUE_MISSING;available="
                + availableDeckIds.Length);
        }
        ClickExact("명령 실행 " + queuedCount + "/" + deckCount,
            "battle-execute");
        yield return null;

        bool battleTerminal = !(bool)P(battle, "HasActiveBattle");
        object stateAfter = P(director, "State");
        object finalization = P(director, "LastTurnFinalization");
        bool finalizationSucceeded = Convert.ToBoolean(P(finalization, "Succeeded"));
        string finalizationFailure = Convert.ToString(P(finalization, "FailureReason"));
        List<object> resolvedCommands = Items(P(director, "LastResolvedTurn"));
        List<object> resolvedEnemyIntents = Items(
            P(director, "LastResolvedEnemyIntents"));
        List<string> resolutions = new List<string>();
        bool executedCommand = false;
        foreach (object resolved in resolvedCommands)
        {
            object execution = F(resolved, "execution");
            string commandOutcome = Convert.ToString(P(execution, "Outcome"));
            executedCommand |= string.Equals(commandOutcome, "Executed",
                StringComparison.Ordinal);
            resolutions.Add("ally:" + Convert.ToString(F(resolved, "characterId"))
                + ":" + commandOutcome + ":effect="
                + P(execution, "AppliedAtLeastOneEffect")
                + ":reason=" + P(execution, "FailureReason")
                + ":target=" + P(execution, "FinalTargetId")
                + ":stages=" + P(F(resolved, "clash"), "AllyStagesRemaining")
                + "/" + P(F(resolved, "clash"), "EnemyStagesRemaining"));
        }
        foreach (object resolved in resolvedEnemyIntents)
        {
            object execution = F(resolved, "execution");
            string commandOutcome = Convert.ToString(P(execution, "Outcome"));
            executedCommand |= string.Equals(commandOutcome, "Executed",
                StringComparison.Ordinal);
            resolutions.Add("enemy:" + Convert.ToString(F(resolved, "enemyId"))
                + ":" + commandOutcome + ":effect="
                + P(execution, "AppliedAtLeastOneEffect")
                + ":reason=" + P(execution, "FailureReason")
                + ":target=" + P(execution, "FinalTargetId")
                + ":stages=" + F(resolved, "requestedExecutionStages"));
        }
        if (!finalizationSucceeded)
        {
            throw new InvalidOperationException(
                "WIM053_BATTLE_TURN_FINALIZATION_FAILED;turn=" + turn
                + ";reason=" + finalizationFailure);
        }
        if (!battleTerminal && (stateAfter == null
                || Convert.ToInt32(F(stateAfter, "turn")) <= turn
                || Items(F(stateAfter, "commandQueue")).Count != 0))
        {
            throw new InvalidOperationException(
                "WIM053_BATTLE_COMMAND_DID_NOT_ADVANCE_ATOMICALLY.");
        }
        float allyAfter = TeamHealth(session, "Allies");
        float enemyAfter = TeamHealth(session, "Enemies");
        long commandAfter = Convert.ToInt64(P(session, "LastProcessedCommandId"));
        float actualEnemyDamage = Mathf.Max(0f, enemyBefore - enemyAfter);
        float actualAllyDamage = Mathf.Max(0f, allyBefore - allyAfter);
        if (executedCommand && commandAfter <= commandBefore)
        {
            throw new InvalidOperationException(
                "WIM053_BATTLE_EXECUTED_COMMAND_ID_DID_NOT_ADVANCE;turn=" + turn
                + ";before=" + commandBefore + ";after=" + commandAfter
                + ";outcomes=" + string.Join(",", resolutions));
        }
        bool objectiveClockAdvanced = string.Equals(
                Convert.ToString(P(P(session, "EncounterRules"), "Objective")),
                "SurviveRounds", StringComparison.Ordinal)
            && Convert.ToInt32(P(session, "RoundNumber")) > roundBefore;
        if (!battleTerminal && actualEnemyDamage <= 0f && actualAllyDamage <= 0f
            && !executedCommand && !objectiveClockAdvanced)
        {
            consecutiveNoEffectBattleTurns++;
        }
        else
        {
            consecutiveNoEffectBattleTurns = 0;
        }
        string battleRow = "turn=" + turn + ";round=" + roundBefore + "->"
            + P(session, "RoundNumber") + ";command=" + commandBefore + "->"
            + commandAfter + ";enemyDamage=" + actualEnemyDamage
            + ";allyDamage=" + actualAllyDamage + ";terminal=" + battleTerminal
            + ";directorFinalized=" + finalizationSucceeded
            + ";finalizeReason=" + finalizationFailure
            + ";outcomes=" + string.Join(",", resolutions)
            + ";deckReadiness=" + deckReadiness
            + ";cards=" + string.Join(",", cards);
        battleRows.Add(battleRow);
        Debug.Log("WIM053_BATTLE_TURN;" + battleRow);
        if (consecutiveNoEffectBattleTurns >= 3)
        {
            throw new InvalidOperationException(
                "WIM053_BATTLE_THREE_NO_EFFECT_TURNS.");
        }
        battleObserved = true;
        battleTurns++;
        enemyDamage += actualEnemyDamage;
    }

    private bool HasAffordableSupplyDebit(object choiceView)
    {
        string choiceId = Convert.ToString(P(choiceView, "ChoiceId"));
        object[] args = { expeditionId, choiceId, null, 0, string.Empty };
        if (!(bool)InvokePublic(decisions, "TryGetActiveChoice", args))
        {
            throw new InvalidOperationException(
                "WIM053_ACTIVE_CHOICE_DEFINITION_UNAVAILABLE;choice="
                + choiceId + ";reason=" + args[4]);
        }
        object expedition = FindActiveExpedition(expeditionId);
        if (expedition == null)
            throw new InvalidOperationException("WIM053_DECISION_EXPEDITION_MISSING.");
        object supplies = P(expedition, "Supplies");
        foreach (object effect in Items(F(args[2], "effects")))
        {
            if (!string.Equals(effect.GetType().Name, "OffenseSupplyDecisionEffect",
                    StringComparison.Ordinal)) continue;
            int amount = Convert.ToInt32(F(effect, "amount"));
            if (amount < 0 && Convert.ToInt32(InvokePublic(supplies,
                    "Get", F(effect, "supplyType"))) >= -amount)
                return true;
        }
        return false;
    }

    private IEnumerator ResolveDecisionThroughUi(object decision)
    {
        string cardId = Convert.ToString(F(decision, "cardId"));
        List<string> rejected = new();
        foreach (object choice in Items(F(decision, "choices"))
                     .OrderByDescending(HasAffordableSupplyDebit))
        {
            string choiceId = Convert.ToString(P(choice, "ChoiceId"));
            string label = Convert.ToString(P(choice, "Label"));
            Button button = ActiveOffenseButtons().FirstOrDefault(value =>
                Label(value).Contains(label, StringComparison.Ordinal));
            if (button == null)
            {
                rejected.Add(choiceId + ":missing");
                continue;
            }
            PointerClick(button, "decision-" + cardId + "-" + choiceId);
            yield return null;
            if (!TryDecision(out _))
            {
                decisionCount++;
                CaptureExpectedConsumption(FindActiveExpedition(expeditionId));
                string row = "card=" + cardId + ";choice=" + choiceId
                    + ";rejected=" + string.Join(",", rejected)
                    + ";consumed=" + string.Join(",", expectedConsumption.Select(pair =>
                        pair.Key + ":" + pair.Value));

            rejected.Add(choiceId + ":not-resolved");
        }
        throw new InvalidOperationException(
            "WIM053_DECISION_NO_PUBLIC_CHOICE_RESOLVED;card=" + cardId
            + ";choices=" + string.Join(",", rejected));
    }

    private static uint DecisionHash(string id, int sequence, int q, int r)
    {
        unchecked
        {
            uint hash = 2166136261u;
            foreach (char value in id ?? string.Empty)
            {
                hash ^= value;
                hash *= 16777619u;
            }
            hash ^= (uint)sequence * 0x9E3779B9u;
            hash ^= (uint)q * 0x85EBCA6Bu;
            hash ^= (uint)r * 0xC2B2AE35u;
            hash ^= hash >> 16;
            return hash;
        }
    }

    private object FindDecisionDetour(object travelState)
    {
        HashSet<string> occupied = new(StringComparer.Ordinal);
        foreach (object site in Items(P(world, "Sites"))
                     .Concat(Items(P(world, "UrgentSites"))))
        {
            if ((bool)P(site, "IsActive"))
            {
                occupied.Add(F(site, "q") + "," + F(site, "r"));
            }
        }
        object dungeon = P(world, "DungeonCoord");
        int dungeonQ = Convert.ToInt32(P(dungeon, "Q"));
        int dungeonR = Convert.ToInt32(P(dungeon, "R"));
        int currentQ = Convert.ToInt32(F(travelState, "currentQ"));
        int currentR = Convert.ToInt32(F(travelState, "currentR"));
        Type coordType = T("OffenseHexCoord", "DungeonStory.Foundation");
        object current = Activator.CreateInstance(coordType,
            new object[] { currentQ, currentR });
        object profile = T("OffenseTravelProfile", "Assembly-CSharp")
            .GetProperty("Default").GetValue(null);
        foreach (object tile in Items(P(world, "Tiles"))
                     .Where(value => !(bool)F(value, "blocked")
                         && !(Convert.ToInt32(F(value, "q")) == dungeonQ
                             && Convert.ToInt32(F(value, "r")) == dungeonR)
                         && !occupied.Contains(F(value, "q") + "," + F(value, "r")))
                     .OrderByDescending(value => HexDistance(currentQ, currentR,
                         Convert.ToInt32(F(value, "q")),
                         Convert.ToInt32(F(value, "r"))))
                     .ThenBy(value => Convert.ToInt32(F(value, "q")))
                     .ThenBy(value => Convert.ToInt32(F(value, "r"))))
        {
            object coord = P(tile, "Coord");
            object[] arguments = { current, coord, profile, null, 0f };
            bool found = (bool)InvokePublic(world, "TryFindPath", arguments);
            List<object> path = found ? Items(arguments[3]) : new List<object>();
            if (path.Count < 2) continue;
            int sequence = Convert.ToInt32(F(travelState, "eventSequence"));
            for (int index = 0; index < path.Count - 1; index++)
            {
                int q = Convert.ToInt32(P(path[index], "Q"));
                int r = Convert.ToInt32(P(path[index], "R"));
                if (DecisionHash(expeditionId, sequence + index, q, r) % 100u < 32u)
                {
                    return coord;
                }
            }
        }
        return null;
    }

    private static int HexDistance(int aq, int ar, int bq, int br)
    {
        int dq = aq - bq;
        int dr = ar - br;
        int ds = -aq - ar - (-bq - br);
        return Math.Max(Math.Abs(dq), Math.Max(Math.Abs(dr), Math.Abs(ds)));
    }

    private static int RewardAuthorityTotal(object rewardState)
    {
        int stock = Items(P(rewardState, "StockGrantedByCategory"))
            .Sum(pair => Convert.ToInt32(P(pair, "Value")));

            + stock
            + CountOf(P(rewardState, "RareFacilityBuildingIds"))
            + CountOf(P(rewardState, "AcquiredBlueprintIds"));
    }

    private IEnumerator VerifySettledResultRoundtrip()
    {
        float previousScale = Time.timeScale;
        Time.timeScale = 0f;
        try
        {
            // Terminal arrival destroys the retired actor at Unity's frame end.
            // Freeze game time but complete that boundary before aggregate capture.
            yield return null;
            object beforeResult = FindResult(expeditionId)
                ?? throw new InvalidOperationException("WIM053_SETTLED_RESULT_MISSING_BEFORE_SAVE.");
            VerifyCurrentValuations();
            string before = ExactSettlementFingerprint();
            string offenseBefore = CaptureOffenseJson();
            int history = CountOf(P(expeditionRuntime, "ResultHistory"));
            int rewards = RewardAuthorityTotal(P(rewardRuntime, "State"));
            int events = rewardEventIds.Count;
            object saveService = Resolve("IDungeonGameSaveService", "DungeonStory.Infrastructure");
            object captured = InvokePublic(saveService, "Capture");
            string json = Convert.ToString(InvokePublic(saveService, "ToJson", captured, false));
            object parsed = InvokePublic(saveService, "FromJson", json);
            object[] arguments = { parsed, null };
            bool restored;
            try { restored = (bool)InvokePublic(saveService, "TryRestore", arguments); }
            finally { Time.timeScale = 0f; }
            if (!restored || arguments[1] == null || !(bool)P(arguments[1], "Success"))
                throw new InvalidOperationException("WIM053_SETTLED_FULLSAVE_RESTORE_FAILED;"
                    + (arguments[1] == null ? "<no-report>" : string.Join("|",
                        Items(P(arguments[1], "Errors")).Select(Convert.ToString))));
            yield return null;
            ResolveJourneyAuthorities();
            object after = FindResult(expeditionId);
            if (after == null
                || !string.Equals(before, ExactSettlementFingerprint(), StringComparison.Ordinal)
                || !string.Equals(offenseBefore, CaptureOffenseJson(), StringComparison.Ordinal)
                || CountOf(P(expeditionRuntime, "ResultHistory")) != history
                || RewardAuthorityTotal(P(rewardRuntime, "State")) != rewards
                || rewardEventIds.Count != events)
                throw new InvalidOperationException("WIM053_SETTLED_RESTORE_CHANGED_RESULT_OR_REGRANTED.");
            object feature = InvokePublic(Resolve("IExpeditionFeatureQueryService"), "Capture");
            string detail = Convert.ToString(InvokePublic(after, "ToDetailText"));
            if (!Items(P(feature, "Results")).Any(row =>
                    string.Equals(Convert.ToString(P(row, "Detail")), detail, StringComparison.Ordinal)))
                throw new InvalidOperationException("WIM053_SETTLED_RESTORE_DETAIL_QUERY_MISMATCH.");
            Debug.Log("WIM053_SETTLED_FULLSAVE_PASS;expedition=" + expeditionId
                + ";jsonChars=" + json.Length + ";history=" + history
                + ";rewardAuthority=" + rewards + "->" + RewardAuthorityTotal(P(rewardRuntime, "State"))
                + ";rewardEvents=" + events + "->" + rewardEventIds.Count
                + ";resultFingerprintEqual=true;detailQueryEqual=true;exactValuationTupleAndOffenseCaptureEqual=true");
        }
        finally { Time.timeScale = previousScale; }
    }

    private void SubscribeBattleCompletion()
    {
        observedBattleSource = battle;
        var completionEvent = battle.GetType().GetEvent("BattleCompleted");
        var callback = GetType().GetMethod("OnObservedBattleCompleted");
        if (completionEvent == null || completionEvent.AddMethod == null || !completionEvent.AddMethod.IsPublic
            || callback == null || !callback.IsPublic)
            throw new InvalidOperationException("WIM053_PUBLIC_BATTLE_COMPLETION_EVENT_MISSING.");
        observedBattleHandler = Delegate.CreateDelegate(completionEvent.EventHandlerType, this,
            callback.MakeGenericMethod(T("OffenseBattleSession", "Assembly-CSharp")));
        completionEvent.AddEventHandler(observedBattleSource, observedBattleHandler);
    }

    public void OnObservedBattleCompleted<TSession>(TSession session)
    {
        if (!string.Equals(Convert.ToString(P(session, "ExpeditionId")), expeditionId, StringComparison.Ordinal)) return;
        completedEnemyIds = Items(P(session, "Combatants"))
            .Where(value => Convert.ToString(P(value, "Team")) == "Enemies")
            .Select(value => Convert.ToString(P(value, "PersistentId")))
            .OrderBy(value => value, StringComparer.Ordinal).ToArray();
        observedBattleOutcome = Convert.ToString(P(session, "Outcome"));
        battleVictoryObserved = string.Equals(observedBattleOutcome, "Victory", StringComparison.Ordinal);
        battleTerminalObserved = true;
        Debug.Log("WIM053_ACTUAL_BATTLE_OUTCOME;expedition=" + expeditionId + ";outcome=" + observedBattleOutcome);
    }

    private void SubscribeRewardEvents()
    {
        object bus = Resolve(
            "DungeonStory.Foundation.IGameEventBus", "DungeonStory.Foundation");
        Type eventType = T("OffenseRewardGrantedEvent", "DungeonStory.Offense");
        var subscribe = bus.GetType().GetMethods().Single(method => method.IsPublic
            && method.IsGenericMethodDefinition
            && string.Equals(method.Name, "Subscribe", StringComparison.Ordinal)
            && method.GetParameters().Length == 1);
        var callback = GetType().GetMethod("OnRewardGranted");
        if (callback == null || !callback.IsPublic || !callback.IsGenericMethodDefinition)
        {
            throw new InvalidOperationException("Public reward callback is unavailable.");
        }
        Delegate handler = Delegate.CreateDelegate(
            typeof(Action<>).MakeGenericType(eventType), this,
            callback.MakeGenericMethod(eventType));
        rewardSubscription = (IDisposable)subscribe.MakeGenericMethod(eventType)
            .Invoke(bus, new object[] { handler });
        if (rewardSubscription == null)
        {
            throw new InvalidOperationException("Reward subscription was not created.");
        }
    }

            ? "<null>"
            : Convert.ToString(P(result, "expeditionId")));
    }

    private void InstallCleanup()
    {
        if (playExitCleanupInstalled) return;
        EditorApplication.playModeStateChanged += OnPlayModeStateChanged;
        playExitCleanupInstalled = true;
    }

    private void Cleanup()
    {
        if (Application.isPlaying) Time.timeScale = 0f;
        EditorApplication.isPaused = true;
        rewardSubscription?.Dispose();
        rewardSubscription = null;
        if (observedBattleSource != null && observedBattleHandler != null)
            observedBattleSource.GetType().GetEvent("BattleCompleted").RemoveEventHandler(
                observedBattleSource, observedBattleHandler);
        observedBattleSource = null;
        observedBattleHandler = null;
        if (playExitCleanupInstalled)
        {
            EditorApplication.playModeStateChanged -= OnPlayModeStateChanged;
            playExitCleanupInstalled = false;
        }
    }

    private void OnPlayModeStateChanged(PlayModeStateChange change)
    {
        if (change == PlayModeStateChange.ExitingPlayMode
            || change == PlayModeStateChange.EnteredEditMode)
        {
            Cleanup();
        }
    }

    public void Execute(ExecutionResult result)
    {
        if (!Application.isPlaying || !EditorApplication.isPlaying
            || !EditorApplication.isPaused || !Mathf.Approximately(Time.timeScale, 0f))
        {
            throw new InvalidOperationException(
                "WIM017053_POST_RESEARCH requires the paused protected prelude.");
        }
        if (EventSystem.current == null || !EventSystem.current.isActiveAndEnabled)
        {
            throw new InvalidOperationException("WIM017053 EventSystem is not ready.");
        }
        InstallCleanup();
        try
        {
            resumedPreparedPlay = SelectPreparedEntry();
            if (resumedPreparedPlay)
            {
                ValidatePreparedResumeResearch();
            }
            else
            {
                CompleteResearchFixture();
            }
            ResolveJourneyAuthorities();
            object ownerManager = FindLoadedComponent("OwnerRunManager");
            object owner = P(ownerManager, "CurrentOwnerActor");
            if (owner == null || !string.Equals(
                    Convert.ToString(P(owner, "CurrentLifecycleState")),
                    "Active", StringComparison.Ordinal))
            {
                throw new InvalidOperationException(
                    "WIM017053_PROTECTED_PARTY_OWNER_MISSING.");
            }
            object campaign = InvokePublic(offenseQuery, "Capture");
            int rationCount = AvailableRations(out int supplyCapacity);
            int runtimeActiveExpeditions = CountOf(
                P(expeditionRuntime, "ActiveExpeditions"));
            if (resumedPreparedPlay
                && (runtimeActiveExpeditions != 0
                    || CountOf(P(campaign, "ActiveExpeditions")) != 0
                    || Mathf.Abs(Convert.ToSingle(P(gameClock, "Time")))
                        > FloatTolerance
                    || rationCount != 24
                    || !(bool)P(campaign, "CanLaunchExpedition")))
            {
                throw new InvalidOperationException(
                    "WIM017053_RESUME_PREPARED_READBACK_MISMATCH;active="
                    + runtimeActiveExpeditions + "/"
                    + CountOf(P(campaign, "ActiveExpeditions"))
                    + ";clock=" + P(gameClock, "Time")
                    + ";rations=" + rationCount
                    + ";unlocked=" + P(campaign, "CanLaunchExpedition"));
            }
            if (!(bool)P(campaign, "IsAvailable")
                || !(bool)P(campaign, "CanLaunchExpedition")
                || CountOf(P(campaign, "ActiveExpeditions")) != 0
                || Convert.ToInt32(P(campaign, "AvailableMemberCount")) < 1
                || rationCount <= 0 || supplyCapacity <= 0)
            {
                throw new InvalidOperationException(
                    "WIM017053_PRODUCTION_READINESS_FAILED;available="
                    + P(campaign, "IsAvailable") + ";canLaunch="
                    + P(campaign, "CanLaunchExpedition") + ";members="
                    + P(campaign, "AvailableMemberCount") + ";rations="
                    + rationCount + ";capacity=" + supplyCapacity);
            }
            EditorApplication.isPaused = false;
            FindScopeHost().StartCoroutine(RunSafely());
            result.Log("WIM017053 POST_RESEARCH installed;entry={0};fixture={1};rations={2};capacity={3};gamePaused={4}",
                resumedPreparedPlay ? "RESUME_PREPARED" : "FRESH_FIXTURE",

    private IEnumerator RunSafely()
    {
        IEnumerator run = Run();
        try
        {
            while (true)
            {
                bool complete = false;
                object current = null;
                Exception failure = null;
                try
                {
                    complete = !run.MoveNext();
                    if (!complete) current = run.Current;
                }
                catch (Exception exception)
                {
                    failure = exception;
                }
                if (failure != null)
                {
                    terminal = "FAILED " + failure.GetBaseException().ToString();
                    Debug.LogError("WIM017053 POST_RESEARCH " + terminal);
                    yield break;
                }
                if (complete) yield break;
                yield return current;
            }
        }
        finally
        {
            (run as IDisposable)?.Dispose();
        }
    }
