    private static object P(object value, string name)
    {
        var property = value?.GetType().GetProperty(name);
        if (property == null || property.GetMethod == null
            || !property.GetMethod.IsPublic)
        {
            throw new InvalidOperationException("Missing public property: " + name);
        }
        return property.GetValue(value);
    }

    private static object PublicStaticField(Type type, string name)
    {

        if (field == null || !field.IsPublic || !field.IsStatic)
        {

                "Missing public static field: " + type.FullName + "." + name);
        }
        return field.GetValue(null);
    }

    private static object InvokePublicNoArgument(object value, string name)
    {
        var method = value?.GetType().GetMethod(name, Type.EmptyTypes);
        if (method == null || !method.IsPublic)
        {
            throw new InvalidOperationException(
                "Missing public no-argument method: " + name);
        }
        return method.Invoke(value, null);
    }

    private static object InvokePublic(object value, string name, object[] arguments)
    {
        var candidates = value?.GetType().GetMethods()
            .Where(method => method.IsPublic && !method.IsStatic
                && string.Equals(method.Name, name, StringComparison.Ordinal))
            .Where(method => method.GetParameters().Length == arguments.Length)
            .ToArray() ?? Array.Empty<System.Reflection.MethodInfo>();
        if (candidates.Length != 1)
        {
            throw new InvalidOperationException(
                "Expected one public " + name + " overload; count="
                + candidates.Length + ".");
        }
        return candidates[0].Invoke(value, arguments);
    }

    private static void Click(Button button)
    {

            || EventSystem.current == null)

            button.gameObject, pointer, ExecuteEvents.pointerEnterHandler);
        ExecuteEvents.Execute(
            button.gameObject, pointer, ExecuteEvents.pointerDownHandler);

            button.gameObject, pointer, ExecuteEvents.pointerUpHandler);

            button.gameObject, pointer, ExecuteEvents.pointerClickHandler);
    }

    private static string FormatStatSnapshot(object stats)

    private IEnumerator DisableSelectedNonResearchPriorities()
    {
        string researchId = IdValue(ResearchWorkTypeId());
        string operateId = IdValue(OperateWorkTypeId());
        foreach (object definition in AllStaffPriorityWorkDefinitions())
        {
            object workTypeId = P(definition, "WorkTypeId");
            string workTypeText = IdValue(workTypeId);
            if (string.Equals(workTypeText, researchId, StringComparison.Ordinal))
            {
                continue;
            }

            string before = WorkPriority(workerWork, workTypeId);
            string after = before;
            int clicks = 0;
            string cellName = "Cell_" + ((UnityEngine.Object)worker).GetInstanceID()
                + "_" + LegacyTypeNameForWorkDefinition(definition);
            while (!string.Equals(after, "Off", StringComparison.Ordinal)
                && clicks < 4)
            {
                Click(FindActiveButton(cellName));
                clicks++;
                selectedNonResearchPriorityClicks++;
                yield return null;
                Canvas.ForceUpdateCanvases();
                after = WorkPriority(workerWork, workTypeId);
            }

            selectedNonResearchPriorityTransitions.Add(workTypeText + ":"
                + before + "->" + after + "/clicks=" + clicks);
            if (!string.Equals(after, "Off", StringComparison.Ordinal))
            {
                throw new InvalidOperationException(
                    "Selected worker non-Research priority did not reach Off through "
                    + "the actual Staff UI: " + workTypeText + ".");
            }

            if (string.Equals(workTypeText, operateId, StringComparison.Ordinal))
            {
                operatePriorityBefore = before;
                operatePriorityAfter = after;
                operatePriorityClicks = clicks;
            }
        }

        if (!string.Equals(
                ResearchPriority(workerWork), "Priority1", StringComparison.Ordinal)
            || !ResearchEnabled(workerWork))
        {
            throw new InvalidOperationException(
                "Selected worker lost Research Priority1 while disabling non-Research UI cells.");
        }
    }

    private static Component[] ActiveResearchPriorityUiActors(
        out string coverage)
    {
        Type actorType = T("CharacterActor", "Assembly-CSharp");
        Component[] loaded = Resources.FindObjectsOfTypeAll(actorType)
            .Cast<UnityEngine.Object>()
            .OfType<Component>()
            .Where(value => value.gameObject.scene.isLoaded
                && value.gameObject.activeInHierarchy)
            .OrderBy(ActorId, StringComparer.Ordinal)
            .ToArray();
        Component owner = loaded.SingleOrDefault(value => (bool)P(value, "IsOwner"));
        if (owner == null)
        {
            throw new InvalidOperationException(
                "R01 PREPARATION_BLOCKER_ACTIVE_OWNER_MISSING_FROM_RESEARCH_PRIORITY_UI.");
        }

        var eligible = new List<Component>();
        var states = new List<string>();
        foreach (Component actor in loaded)
        {
            string actorId = ActorId(actor);
            bool ownerActor = (bool)P(actor, "IsOwner");
            bool dead = (bool)P(actor, "IsDead");
            object work = null;
            bool hasWork = !dead && TryGetPublicWork(actor, out work);
            if (!hasWork)
            {
                states.Add(actorId + ";owner=" + ownerActor + ";dead=" + dead
                    + ";workRole=false;researchCapable=false");
                continue;
            }

            Button cell = RequireActiveResearchPriorityCell(actor);
            if (ownerActor && ResearchEnabled(work) && !cell.IsInteractable())
            {
                throw new InvalidOperationException(
                    "R01 PREPARATION_BLOCKER_OWNER_RESEARCH_PRIORITY_UI_NOT_INTERACTABLE; "
                    + "actor=" + actorId + ";cell=" + cell.name + ".");
            }

            eligible.Add(actor);
            states.Add(actorId + ";owner=" + ownerActor + ";dead=false"
                + ";workRole=true;research=" + ResearchPriority(work)
                + ";cell=" + cell.name);
        }

        if (eligible.Count == 0)
        {
            throw new InvalidOperationException(
                "R01 PREPARATION_BLOCKER_RESEARCH_PRIORITY_UI_HAS_NO_ACTIVE_WORKERS.");
        }

        coverage = string.Join("|", states);
        return eligible.OrderBy(ActorId, StringComparer.Ordinal).ToArray();
    }

    private IEnumerator WarmResearchFacilityCandidateIndex()
    {
        object world = Resolve("ICharacterAiWorldRegistry", "Assembly-CSharp");
        Type gridType = T("Grid", "DungeonStory.Grid");
        var tryGetGrid = world.GetType().GetMethod(
            "TryGetGrid", new[] { gridType.MakeByRefType() });
        if (tryGetGrid == null || !tryGetGrid.IsPublic)
        {
            throw new InvalidOperationException(
                "Public ICharacterAiWorldRegistry.TryGetGrid is unavailable for index warmup.");
        }

        object[] gridArguments = { null };
        if (!(bool)tryGetGrid.Invoke(world, gridArguments) || gridArguments[0] == null)
        {
            throw new InvalidOperationException(
                "R01 PREPARATION_BLOCKER_LIVE_GRID_MISSING_FOR_RESEARCH_INDEX_WARMUP.");
        }

        object cache = Resolve("IFacilityCandidateCache", "Assembly-CSharp");
        Type facilityWorkType = T("FacilityWorkType", "DungeonStory.Buildings");
        var getWorkCandidates = cache.GetType().GetMethod(
            "GetWorkCandidates", new[] { gridType, facilityWorkType });
        if (getWorkCandidates == null || !getWorkCandidates.IsPublic)
        {
            throw new InvalidOperationException(
                "Public IFacilityCandidateCache.GetWorkCandidates is unavailable for index warmup.");
        }

        object research = Enum.Parse(facilityWorkType, "Research");
        int queryCount = 0;
        getWorkCandidates.Invoke(cache, new[] { gridArguments[0], research });
        queryCount++;
        bool pendingBefore = (bool)P(cache, "HasPendingIndexBuild");
        int versionBefore = Convert.ToInt32(P(cache, "CandidateIndexVersion"));
        if (!pendingBefore)
        {
            researchIndexWarmup = "pendingBefore=false;queries=" + queryCount
                + ";indexVersion=" + versionBefore + ";normalSchedulerFrames=0";
            yield break;
        }

        object warmupClock = Resolve(
            "DungeonStory.Foundation.IGameClock",
            "DungeonStory.Foundation");
        float gameStart = Convert.ToSingle(P(warmupClock, "Time"));
        float wallStart = Time.realtimeSinceStartup;
        int schedulerFrames = 0;
        researchIndexWarmup = "pendingBefore=true;queries=" + queryCount
            + ";indexVersionBefore=" + versionBefore + ";state=warming";
        try
        {
            Time.timeScale = 1f;
            yield return null;
            schedulerFrames++;
            if ((bool)P(warmupClock, "IsPaused"))
            {
                throw new InvalidOperationException(
                    "R01 PREPARATION_BLOCKER_RESEARCH_INDEX_SCHEDULER_PAUSED.");
            }

            while ((bool)P(cache, "HasPendingIndexBuild"))
            {
                float elapsedGame = Convert.ToSingle(P(warmupClock, "Time"))
                    - gameStart;
                float elapsedWall = Time.realtimeSinceStartup - wallStart;
                if (elapsedGame >= IndexWarmupMaximumGameSeconds
                    || elapsedWall >= IndexWarmupMaximumWallSeconds)
                {
                    throw new InvalidOperationException(
                        "R01 PREPARATION_BLOCKER_RESEARCH_INDEX_WARMUP_TIMEOUT; "
                        + "game=" + elapsedGame + ";wall=" + elapsedWall
                        + ";indexVersion=" + Convert.ToString(
                            P(cache, "CandidateIndexVersion")));
                }

                // This public query triggers the known-grid index but does not advance it.
                getWorkCandidates.Invoke(cache, new[] { gridArguments[0], research });
                queryCount++;
                yield return null;
                schedulerFrames++;
            }
        }
        finally
        {
            Time.timeScale = 0f;
        }

        // Do not retain a list captured before the scheduler completed the index.
        getWorkCandidates.Invoke(cache, new[] { gridArguments[0], research });
        queryCount++;
        if ((bool)P(cache, "HasPendingIndexBuild"))
        {
            throw new InvalidOperationException(
                "R01 PREPARATION_BLOCKER_RESEARCH_INDEX_STILL_PENDING_AFTER_WARMUP.");
        }

        researchIndexWarmup = "pendingBefore=true;queries=" + queryCount
            + ";normalSchedulerFrames=" + schedulerFrames
            + ";game=" + gameStart + "->"
            + Convert.ToSingle(P(warmupClock, "Time"))
            + ";wall=" + (Time.realtimeSinceStartup - wallStart)
            + ";indexVersion=" + Convert.ToString(
                P(cache, "CandidateIndexVersion"))
            + ";paused=" + Mathf.Approximately(Time.timeScale, 0f);
    }

    private static object FindPriorityWorker(
        IEnumerable<Component> researchPriorityActors,
        out object work)
    {
        foreach (Component actor in researchPriorityActors
                     .Where(value => !(bool)P(value, "IsOwner")))
        {
            if (!TryGetPublicWork(actor, out object candidateWork))
            {
                continue;
            }
            string cellName = "Cell_" + actor.GetInstanceID() + "_Research";
            Button cell = Resources.FindObjectsOfTypeAll<Button>()
                .SingleOrDefault(value => value != null
                    && value.gameObject.scene.isLoaded
                    && value.gameObject.activeInHierarchy
                    && string.Equals(value.name, cellName, StringComparison.Ordinal));
            if (cell != null)
            {
                work = candidateWork;
                return actor;
            }
        }
        work = null;
        throw new InvalidOperationException(
            "No active living staff actor has a visible Research priority cell.");
    }

    private IEnumerator RequireReachableResearchFacility(float wallDeadline)
    {
        if (!(bool)P(runtime, "HasActiveResearch"))
        {
            throw new InvalidOperationException(
                "R01 PREPARATION_BLOCKER_ACTIVE_RESEARCH_MISSING after actual queue readback.");
        }

        object world = Resolve("ICharacterAiWorldRegistry", "Assembly-CSharp");
        Type gridType = T("Grid", "DungeonStory.Grid");
        var tryGetGrid = world.GetType().GetMethod(
            "TryGetGrid", new[] { gridType.MakeByRefType() });
        if (tryGetGrid == null || !tryGetGrid.IsPublic)
        {
            throw new InvalidOperationException(
                "Public ICharacterAiWorldRegistry.TryGetGrid is unavailable.");
        }
        object[] gridArguments = { null };
        if (!(bool)tryGetGrid.Invoke(world, gridArguments) || gridArguments[0] == null)
        {
            throw new InvalidOperationException(
                "R01 PREPARATION_BLOCKER_LIVE_GRID_MISSING.");
        }

        object cache = Resolve("IFacilityCandidateCache", "Assembly-CSharp");
        Type facilityWorkType = T("FacilityWorkType", "DungeonStory.Buildings");
        var getWorkCandidates = cache.GetType().GetMethod(
            "GetWorkCandidates", new[] { gridType, facilityWorkType });
        if (getWorkCandidates == null || !getWorkCandidates.IsPublic)
        {
            throw new InvalidOperationException(
                "Public IFacilityCandidateCache.GetWorkCandidates is unavailable.");
        }
        object indexed = getWorkCandidates.Invoke(cache, new[]
        {
            gridArguments[0],
            Enum.Parse(facilityWorkType, "Research")
        });
        if ((bool)P(cache, "HasPendingIndexBuild"))
        {
            throw new InvalidOperationException(
                "R01 PREPARATION_BLOCKER_RESEARCH_INDEX_PENDING; indexVersion="
                + Convert.ToString(P(cache, "CandidateIndexVersion")));
        }

        object policy = Resolve("IWorkPolicyRegistry", "Assembly-CSharp");
        Type workTypeId = T("WorkTypeId", "DungeonStory.Work");
        Type actorType = T("CharacterActor", "Assembly-CSharp");
        Type buildingType = T("BuildableObject", "Assembly-CSharp");
        var isAvailable = policy.GetType().GetMethod("IsAvailable", new[]
        {
            workTypeId,
            actorType,
            buildingType,
            typeof(string).MakeByRefType()
        });
        if (isAvailable == null || !isAvailable.IsPublic)
        {
            throw new InvalidOperationException(
                "Public IWorkPolicyRegistry.IsAvailable is unavailable.");
        }

        var supportsWork = buildingType.GetMethod(
            "SupportsWork", new[] { workTypeId });
        var access = buildingType.GetMethod("TryGetNearestWorkAccessGridPosition", new[]
        {
            gridType,
            typeof(Vector2Int),
            typeof(Vector2Int).MakeByRefType()
        });
        if (supportsWork == null || !supportsWork.IsPublic
            || access == null || !access.IsPublic)
        {
            throw new InvalidOperationException(
                "Public research facility capability/access query is unavailable.");
        }

        List<string> rejected = new();
        foreach (object facility in ((IEnumerable)indexed).Cast<object>()
                     .Where(value => value is Component component
                         && component.gameObject.scene.isLoaded
                         && component.gameObject.activeInHierarchy)
                     .OrderBy(value => ((UnityEngine.Object)value).GetInstanceID()))
        {
            string facilityLabel = ((Component)facility).name + "#"
                + ((UnityEngine.Object)facility).GetInstanceID();
            bool indexedSupportsResearch = (bool)supportsWork.Invoke(
                facility, new[] { ResearchWorkTypeId() });
            if (!indexedSupportsResearch)
            {
                rejected.Add(facilityLabel + ":indexed-without-support");
                continue;
            }

            object[] availabilityArguments =
            {
                ResearchWorkTypeId(), worker, facility, null
            };
            bool workforceAvailable = (bool)isAvailable.Invoke(
                policy, availabilityArguments);
            if (!workforceAvailable)
            {
                rejected.Add(facilityLabel + ":workforce="
                    + Convert.ToString(availabilityArguments[3]));
                continue;
            }

            object[] accessArguments =
            {
                gridArguments[0],
                (Vector2Int)InvokePublicNoArgument(worker, "GetNowXY"),
                default(Vector2Int)
            };
            if (!(bool)access.Invoke(facility, accessArguments))
            {
                rejected.Add(facilityLabel + ":no-work-access");
                continue;
            }

            string pathStatus = RequestActorReachability(
                worker,
                gridArguments[0],
                (Vector2Int)accessArguments[2]);
            while (string.Equals(pathStatus, "Pending", StringComparison.Ordinal))
            {
                if (Time.realtimeSinceStartup >= wallDeadline)
                {
                    throw new InvalidOperationException(
                        "R01 PREPARATION_BLOCKER_RESEARCH_PATH_PENDING; facility="
                        + facilityLabel + ".");
                }
                yield return null;
                pathStatus = RequestActorReachability(
                    worker,
                    gridArguments[0],
                    (Vector2Int)accessArguments[2]);
            }
            if (!string.Equals(pathStatus, "Reachable", StringComparison.Ordinal))
            {
                rejected.Add(facilityLabel + ":path=" + pathStatus);
                continue;
            }

            researchFacilityPreflight = "activeProject=" + expectedProjectId
                + ";indexVersion=" + Convert.ToString(P(
                    cache, "CandidateIndexVersion"))
                + ";facility=" + facilityLabel
                + ";indexedSupportsResearch=true;workforce=true;path="
                + pathStatus;
            yield break;
        }

        researchFacilityPreflight = "activeProject=" + expectedProjectId
            + ";indexVersion=" + Convert.ToString(P(
                cache, "CandidateIndexVersion"))
            + ";rejected=" + (rejected.Count == 0
                ? "<no-indexed-research-facility>"
                : string.Join("|", rejected));
        throw new InvalidOperationException(
            "R01 PREPARATION_BLOCKER_REACHABLE_RESEARCH_FACILITY_MISSING; "
            + researchFacilityPreflight);
    }

    private void RequireHydrationAvailability()
    {
        object items = Resolve("IWorldItemStackRuntime", "Assembly-CSharp");
        int before = GetAvailableSafeCleanWater(items, out string beforeDetail);
        int missing = Mathf.Max(0, CurrentThreePersonOneDayWaterUnits - before);
        int spawned = missing > 0
            ? SpawnMissingCleanWaterAtDeliveryDropoff(items, missing)
            : 0;
        int after = GetAvailableSafeCleanWater(items, out string afterDetail);
        hydrationAvailability = "safeCleanWaterBefore=" + before
            + ";required=" + CurrentThreePersonOneDayWaterUnits
            + ";spawnedAtDeliveryDropoff=" + spawned
            + ";safeCleanWaterAfter=" + after
            + ";before=" + beforeDetail
            + ";after=" + afterDetail;
        if (after < CurrentThreePersonOneDayWaterUnits)
        {
            throw new InvalidOperationException(
                "R01 PREPARATION_BLOCKER_SAFE_CLEAN_WATER_UNAVAILABLE; "
                + hydrationAvailability);
        }
    }

    private static int GetAvailableSafeCleanWater(
        object items,
        out string detail)
    {
        Type stockCategory = T("StockCategory", "DungeonStory.Buildings");
        Type candidateType = T("WorldItemStockCandidate", "Assembly-CSharp");
        Type candidateListType = typeof(List<>).MakeGenericType(candidateType);
        var copyCandidates = items.GetType().GetMethod(
            "CopyAvailableStockCandidates",
            new[] { stockCategory, candidateListType });
        if (copyCandidates == null || !copyCandidates.IsPublic)
        {
            throw new InvalidOperationException(
                "Public IWorldItemStackRuntime.CopyAvailableStockCandidates is unavailable.");
        }

        object candidates = Activator.CreateInstance(candidateListType);
        copyCandidates.Invoke(items, new[]
        {
            Enum.Parse(stockCategory, "Water"),
            candidates
        });
        var stacksById = ((IEnumerable)InvokePublicNoArgument(items, "GetAllStacks"))
            .Cast<object>()
            .Where(value => value != null)
            .ToDictionary(
                value => Convert.ToString(P(value, "StackId")),
                StringComparer.Ordinal);
        int available = 0;
        var accepted = new List<string>();
        foreach (object candidate in (IEnumerable)candidates)
        {
            if (!(bool)P(candidate, "IsValid"))
            {
                continue;
            }

            string stackId = Convert.ToString(P(candidate, "StackId"));
            if (!stacksById.TryGetValue(stackId, out object stack)
                || !string.Equals(
                    Convert.ToString(P(stack, "ItemId")),
                    CleanWaterItemId,
                    StringComparison.Ordinal))
            {
                continue;
            }

            int stackAvailable = Convert.ToInt32(P(stack, "AvailableQuantity"));
            if (stackAvailable <= 0)
            {
                continue;
            }

            available += stackAvailable;
            accepted.Add(stackId + "@" + Convert.ToString(P(candidate, "Position"))
                + ":" + Convert.ToString(P(candidate, "State"))
                + ":available=" + stackAvailable);
        }

        detail = accepted.Count == 0
            ? "<no-safe-resource:clean-water-candidate>"
            : string.Join("|", accepted);
        return available;
    }

    private static int SpawnMissingCleanWaterAtDeliveryDropoff(
        object items,
        int missing)
    {
        object dropZones = Resolve("IWorldDropZoneQuery", "Assembly-CSharp");
        var getDropoff = dropZones.GetType().GetMethod(
            "TryGetDeliveryDropoff",
            new[] { typeof(Vector2Int).MakeByRefType() });
        if (getDropoff == null || !getDropoff.IsPublic)
        {
            throw new InvalidOperationException(
                "Public IWorldDropZoneQuery.TryGetDeliveryDropoff is unavailable.");
        }

        object[] dropoffArguments = { default(Vector2Int) };
        if (!(bool)getDropoff.Invoke(dropZones, dropoffArguments))
        {
            throw new InvalidOperationException(
                "R01 PREPARATION_BLOCKER_DELIVERY_DROPOFF_MISSING; "
                + "clean-water physical setup was not attempted.");
        }

        Type itemState = T("WorldItemStackState", "DungeonStory.Items");
        var spawn = items.GetType().GetMethod("SpawnItemAt", new[]
        {
            typeof(string),
            typeof(int),
            typeof(Vector2Int),
            itemState,
            typeof(string),
            typeof(int).MakeByRefType()
        });
        if (spawn == null || !spawn.IsPublic)
        {
            throw new InvalidOperationException(
                "Public IWorldItemStackRuntime.SpawnItemAt is unavailable.");
        }

        object[] spawnArguments =
        {
            CleanWaterItemId,
            missing,
            (Vector2Int)dropoffArguments[0],
            Enum.Parse(itemState, "Loose"),
            string.Empty,
            0
        };
        bool accepted = (bool)spawn.Invoke(items, spawnArguments);
        int spawned = Convert.ToInt32(spawnArguments[5]);
        if (!accepted || spawned != missing)
        {
            throw new InvalidOperationException(
                "R01 PREPARATION_BLOCKER_PHYSICAL_CLEAN_WATER_SPAWN_REJECTED; "
                + "requested=" + missing + ";spawned=" + spawned
                + ";dropoff=" + Convert.ToString(dropoffArguments[0]));
        }

        return spawned;
    }

    private static string ExtractDetailLine(object researchWindow, string token)
    {
        Component window = researchWindow as Component;
        Type textType = T("TMPro.TMP_Text", "Unity.TextMeshPro");
        string[] lines = window.GetComponentsInChildren(textType, true)
            .Cast<Component>()
            .Select(value => Convert.ToString(P(value, "text")))
            .Where(value => !string.IsNullOrWhiteSpace(value))
            .SelectMany(value => value.Split('\n'))
            .Where(value => value.Contains(token, StringComparison.Ordinal))
            .Distinct(StringComparer.Ordinal)
            .ToArray();
        if (lines.Length != 1)
        {
            throw new InvalidOperationException(
                "Expected one Research UI detail line for " + token
                + "; count=" + lines.Length);
        }
        return lines[0].Trim();
    }

    private void SubscribeResearchProgress()
    {
        object eventBus = Resolve(
            "DungeonStory.Foundation.IGameEventBus",
            "DungeonStory.Foundation");
        Type eventType = T("ResearchProgressEvent", "Assembly-CSharp");
        var subscribe = eventBus.GetType().GetMethods()
            .Single(value => value.Name == "Subscribe"
                && value.IsPublic
                && value.IsGenericMethodDefinition
                && value.GetParameters().Length == 1);
        var callbackMethod = GetType().GetMethod("OnResearchProgress");
        if (callbackMethod == null || !callbackMethod.IsPublic
            || !callbackMethod.IsGenericMethodDefinition)
        {
            throw new InvalidOperationException(
                "Public verifier callback method is unavailable.");
        }
        var closedCallback = callbackMethod.MakeGenericMethod(eventType);
        Type actionType = typeof(Action<>).MakeGenericType(eventType);
        Delegate callback = Delegate.CreateDelegate(
            actionType, this, closedCallback);
        progressSubscription = (IDisposable)subscribe
            .MakeGenericMethod(eventType)
            .Invoke(eventBus, new object[] { callback });
        if (progressSubscription == null)
        {
            throw new InvalidOperationException(
                "ResearchProgressEvent subscription was not created.");
        }
    }

    public void OnResearchProgress<TEvent>(TEvent value)
    {
        object gameEvent = value;
        string eventProjectId = Convert.ToString(P(gameEvent, "ProjectId"));
        string eventActorId = IdValue(P(gameEvent, "Researcher"));
        float approved = Convert.ToSingle(P(gameEvent, "ApprovedWork"));
        float delta = Convert.ToSingle(P(gameEvent, "ProgressDelta"));
        int day = Convert.ToInt32(P(gameEvent, "AbsoluteDay"));
        researchProgressEventRecords.Add("actor=" + eventActorId
            + ";project=" + eventProjectId
            + ";approved=" + approved
            + ";delta=" + delta
            + ";day=" + day);
        if (!string.Equals(
                eventProjectId, expectedProjectId, StringComparison.Ordinal)
            || approved <= 0f || delta <= 0f)
        {
            return;
        }
        if (!string.Equals(eventActorId, expectedActorId, StringComparison.Ordinal))
        {
            unexpectedContributor = true;
            return;
        }
        matchingEventCount++;
        approvedWorkSum += approved;
        progressDeltaSum += delta;
        lastAbsoluteDay = day;
    }

    private void InstallPlayExitCleanup()
    {
        if (playModeCallbackInstalled)
        {
            return;
        }
        EditorApplication.playModeStateChanged += OnPlayModeStateChanged;
        playModeCallbackInstalled = true;
    }

    private void CleanupCallbacks()
    {
        Exception cleanupFailure = null;
        try
        {
            if (Application.isPlaying)
            {
                Time.timeScale = 0f;
            }
            EditorApplication.isPaused = true;
        }
        catch (Exception exception)
        {
            cleanupFailure = exception;
        }
        try
        {
            progressSubscription?.Dispose();
        }
        catch (Exception exception)
        {
            cleanupFailure ??= exception;
        }
        finally
        {
            progressSubscription = null;
        }
        try
        {
            if (playModeCallbackInstalled)
            {
                EditorApplication.playModeStateChanged -= OnPlayModeStateChanged;
            }
        }
        catch (Exception exception)
        {
            cleanupFailure ??= exception;
        }
        finally
        {
            playModeCallbackInstalled = false;
        }
        if (cleanupFailure != null)
        {
            throw new InvalidOperationException(
                "R01 callback cleanup failed after attempting pause/dispose/unsubscribe.",
                cleanupFailure);
        }
    }

    private void OnPlayModeStateChanged(PlayModeStateChange state)
    {
        if (state != PlayModeStateChange.ExitingPlayMode
            && state != PlayModeStateChange.EnteredEditMode)
        {
            return;
        }
        bool hadCallbacks = playModeCallbackInstalled
            || progressSubscription != null;
        CleanupCallbacks();
        if (hadCallbacks)
        {
            Debug.Log("R01 ONE_WORKER PLAY_EXIT_CLEANUP");
        }
    }

    public void Execute(ExecutionResult result)
    {
        if (!EditorApplication.isPlaying || !Application.isPlaying
            || !EditorApplication.isPaused
            || !Mathf.Approximately(Time.timeScale, 0f))
        {
            throw new InvalidOperationException(
                "R01 one-worker command requires the paused protected Play prelude.");
        }
        InstallPlayExitCleanup();
        try
        {
            // Frames must run for public UI/path requests, but game time remains paused
            // until the bounded normal-AI interval below explicitly starts it.
            EditorApplication.isPaused = false;
            FindRuntimeScopeHost().StartCoroutine(RunSafely());
        }
        catch
        {
            CleanupCallbacks();
            throw;
        }
        result.Log("R01 ONE_WORKER installed;gamePaused={0};editorPaused={1};speed=normal-1",
            Mathf.Approximately(Time.timeScale, 0f), EditorApplication.isPaused);
    }

    private IEnumerator RunSafely()
    {
        IEnumerator run = Run();
        try
        {
            while (true)
            {
                object next = null;
                Exception runFailure = null;
                bool completed = false;
                try
                {
                    if (!run.MoveNext())
                    {
                        completed = true;
                    }
                    else
                    {
                        next = run.Current;
                    }
                }
                catch (Exception exception)
                {
                    runFailure = exception;
                }
                if (runFailure != null)
                {
                    Debug.LogError(
                        "R01 ONE_WORKER FAILED;detail="
                        + runFailure.GetType().Name + ": "
                        + runFailure.Message);
                    yield break;
                }
                if (completed)
                {
                    yield break;
                }
                yield return next;
            }
        }
        finally
        {
            (run as IDisposable)?.Dispose();
        }
    }

    private IEnumerator Run()
    {
        bool passed = false;
        bool incomplete = false;
        string outcome = "FAILED before public readiness gates";
        try
        {
            runtime = FindLoadedComponent(
                "BlueprintResearchRuntime", "Assembly-CSharp");
            expectedProjectReadiness = "runtime=loaded;catalog=loaded;expected="
                + ExpectedProjectId;
            project = FindExpectedProject(runtime);
            projectId = P(project, "ProjectId");
            expectedProjectId = ProjectIdOf(project);
            requiredWork = Convert.ToSingle(P(project, "RequiredWork"));
            maximumResearchers = Convert.ToInt32(P(project, "MaximumResearchers"));
            string expectedNodeState = NodeState(runtime, project);
            string[] prerequisiteIds = ((IEnumerable)P(project, "Prerequisites"))
                .Cast<object>()
                .Where(value => value != null)
                .Select(ProjectIdOf)
                .OrderBy(value => value, StringComparer.Ordinal)
                .ToArray();
            expectedProjectReadiness = "id=" + expectedProjectId
                + ";state=" + expectedNodeState
                + ";required=" + requiredWork
                + ";maxResearchers=" + maximumResearchers
                + ";prerequisites="
                + (prerequisiteIds.Length == 0
                    ? "<none>"
                    : string.Join(",", prerequisiteIds));
            if (!string.Equals(
                    expectedNodeState, "Available", StringComparison.Ordinal)
                || !string.Equals(
                    expectedProjectId, ExpectedProjectId, StringComparison.Ordinal)
                || !Mathf.Approximately(requiredWork, ExpectedRequiredWork)
                || maximumResearchers != 1
                || prerequisiteIds.Length != 0)
            {
                throw new InvalidOperationException(
                    "Expected research project readiness mismatch: "
                    + expectedProjectReadiness);
            }

            object capacity = P(runtime, "ResearchFacilityCapacity");
            Type capabilityType = T(
                "ResearchFacilityCapabilityId", "DungeonStory.Research");
            object basic = Enum.Parse(capabilityType, "Basic");
            var getAvailable = capacity.GetType().GetMethod(
                "GetAvailable", new[] { capabilityType });
            if (getAvailable == null || !getAvailable.IsPublic)
            {
                throw new InvalidOperationException(
                    "Public research facility capacity query is unavailable.");
            }
            basicCapacity = Convert.ToInt32(
                getAvailable.Invoke(capacity, new[] { basic }));
            var meetsRequirements = capacity.GetType().GetMethods()
                .Single(value => value.Name == "MeetsRequirements"
                    && value.IsPublic
                    && value.GetParameters().Length == 2
                    && value.GetParameters()[0].ParameterType
                        == T("ResearchProjectSO", "DungeonStory.Research")
                    && value.GetParameters()[1].ParameterType
                        == typeof(string).MakeByRefType());
            object[] capacityArguments = { project, null };
            bool facilitiesReady = (bool)meetsRequirements.Invoke(
                capacity, capacityArguments);
            expectedProjectReadiness += ";basicCapacity=" + basicCapacity
                + ";facilitiesReady=" + facilitiesReady
                + ";facilityBlocker=" + Convert.ToString(capacityArguments[1]);
            if (basicCapacity != 1 || !facilitiesReady)
            {
                throw new InvalidOperationException(
                    "Research facility readiness mismatch: Basic="
                    + basicCapacity + ";blocker="
                    + Convert.ToString(capacityArguments[1]));
            }

            object projectsState = ProjectsState(runtime);
            bool expectedCompleted = PublicProjectBool(
                projectsState, "IsCompleted", projectId);
            bool expectedQueued = PublicProjectBool(
                projectsState, "ContainsInQueue", projectId);
            string[] queueIds = QueueIds(projectsState);
            string activeProjectBeforeWarmup = IdValue(P(
                projectsState, "ActiveProjectId"));
            expectedProjectReadiness += ";completed=" + expectedCompleted
                + ";expectedQueued=" + expectedQueued
                + ";queueCount=" + queueIds.Length
                + ";activeBeforeWarmup=" + activeProjectBeforeWarmup;
            if (expectedCompleted || expectedQueued || queueIds.Length != 0
                || !string.IsNullOrWhiteSpace(activeProjectBeforeWarmup))
            {
                throw new InvalidOperationException(

## Stop / evidence contract

- Run only after the accepted protected prelude has disposed persistence, reconciled TierZero, created the actual party through Owner/Confirm UI, and left time paused. The public catalog must contain exact `research:commerce:logistics`; its own public node state must be Available, its prerequisite list empty, RequiredWork 17, MaximumResearchers 1, Basic capacity exactly 1, facility requirements met, and both the project queue and ActiveProjectId empty. Catalog enumeration order and any other Available root are diagnostic only, never selection authority.
- Before the root is queued or the scheduler warmup begins, the command opens the actual Staff priority panel through public `UITabButtonBinding.Id == TabId.Staff` -> `P1Action_StaffModePriorities`. Its production model includes every valid, living actor with an `AbilityWork` role, including an owner (`StaffWorkforceRuntimeQueryService.FindActiveWorkers` has no owner exclusion); this command requires each current research-capable actor's exact live `Cell_{instanceId}_Research` row. A work-capable owner without that row, or an enabled owner row that is not interactable, is a named preparation blocker rather than a game/UI change.
- The selected non-owner remains the benchmark worker and is set to Research Priority1 through its live cell. Every other enabled Research row, including the owner when capable, is clicked to Off, and the count must then be exactly that selected actor across all current rows. The selected actor then has every non-Research `WorkTaskCatalog.All` work type mapped through `FacilityWorkTypeMap.GetRequired` and clicked through its own live Staff cell until Off; ordinary deprivation/self-care commands are never disabled, set, or invoked. This is a dedicated one-worker research benchmark, not a daily-life production-rate measurement, and all row/priority transitions are reported.
- Only after that isolation, the command triggers the known grid through public `IFacilityCandidateCache.GetWorkCandidates(grid, FacilityWorkType.Research)` and reads public `HasPendingIndexBuild`. Only when pending, it permits the normal scheduler to run at time-scale 1 for at most 5 game seconds or 30 wall seconds, then pauses in `finally`; it never calls `AdvanceIndex`, Tick, a work selector, or any force-progress API. A final public `GetWorkCandidates` call is deliberately reacquired after the scheduler completes, so no pre-warmup list is retained.
