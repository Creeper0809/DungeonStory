            && property.GetMethod.IsPublic)
            return property.GetValue(value);
        var field = value.GetType().GetField(name);

        EditorApplication.playModeStateChanged += OnPlayModeStateChanged;
        EditorApplication.isPaused = false;
        phase = 0;
        earliestPlayerFrame = Time.frameCount + 1;
        result.Log(
            "WIM043 INSTALLED autosaveDisposed=true;metaDisposed=true;tierZero={0}->{1};ownerClick=OwnerOption_1001;seed={2};paused={3}",
            before, after, RunSeed, Mathf.Approximately(Time.timeScale, 0f));
    }

    void OnEditorUpdate()
    {
        if (!installed)
            return;
        if (EditorApplication.timeSinceStartup - wallStart >= MaximumWallSeconds)
        {
            Finish("INCOMPLETE_WALL_TIMEOUT", false,
                "Bounded public run did not reach the natural typed-delay expiry.");
            return;
        }
        if (Time.frameCount < earliestPlayerFrame)
            return;
        try
        {
            if (!EditorApplication.isPlaying || !Application.isPlaying)
            {
                Unsubscribe();
                return;
            }
            if (callbackFailure)
                throw new InvalidOperationException(callbackFailureDetail);
            Step();
        }
        catch (Exception exception)
        {
            Finish("FAILED_PHASE_" + phase, true,
                exception.GetType().Name + ": " + exception.Message);
        }
    }

    void Step()
    {
        if (phase == 0)
        {
            Button confirm = FindActiveButton("StartPartyConfirm", false);
            if (confirm == null || !confirm.IsInteractable())
                return;
            CreatePredictAndApply();
            Advance(1);
            return;
        }
        if (phase == 1)
        {
            Component[] loaded = ActiveActors();
            if (loaded.Length != 3
                || FindActiveButton("StartPartyConfirm", false) != null)
                return;
            SetupGameplay(loaded);
            Advance(10);
            return;
        }
        if (phase == 2)
        {
            Canvas.ForceUpdateCanvases();
            Click(FindActiveButton("P1Action_StaffModePriorities"));
            BuildResearchPriorityTasks();
            Advance(3);
            return;
        }
        if (phase == 3)
        {
            if (!ProcessPriorityTask())
                return;
            BeginResearchIndexWarmup();
            Advance(4);
            return;
        }
        if (phase == 4)
        {
            if (!TickResearchIndexWarmup())
                return;
            Time.timeScale = 0f;
            Click(FindActiveButton("TopTabButton_Research_연구"));
            Advance(5);
            return;
        }
        if (phase == 5)
        {
            Canvas.ForceUpdateCanvases();
            Click(FindActiveButton("Node_" + GatheringId));
            Advance(6);
            return;
        }
        if (phase == 6)
        {
            QueueProjectThroughUi(gatheringProject, GatheringId);
            Time.timeScale = 1f;
            Advance(7);
            return;
        }
        if (phase == 7)
        {
            if (!WaitForResearch(gatheringProject, GatheringId))
                return;
            Time.timeScale = 0f;
            Click(FindActiveButton("Node_" + FieldId));
            Advance(8);
            return;
        }
        if (phase == 8)
        {
            QueueProjectThroughUi(fieldProject, FieldId);
            Time.timeScale = 1f;
            Advance(9);
            return;
        }
        if (phase == 9)
        {
            if (!WaitForResearch(fieldProject, FieldId))
                return;
            Time.timeScale = 0f;
            PlaceInitialP23();
            Click(FindActiveButton("TopTabButton_Research_연구"));
            Advance(10);
            return;
        }
        if (phase == 10)
        {
            Click(FindActiveTopTabButton("Staff"));
            Advance(11);
            return;
        }
        if (phase == 11)
        {
            Canvas.ForceUpdateCanvases();
            Click(FindActiveButton("P1Action_StaffModePriorities"));
            BuildSowPriorityTasks();
            Advance(12);
            return;
        }
        if (phase == 12)
        {
            if (!ProcessPriorityTask())
                return;
            DisplayPlot();
            Advance(13);
            return;
        }
        if (phase == 13)
        {
            Canvas.ForceUpdateCanvases();
            Click(FindCropChoiceButton());
            Advance(14);
            return;
        }
        if (phase == 14)
        {
            plotSnapshot = FindPlotSnapshot(cropRuntime, plotId);
            string crop = Convert.ToString(V(plotSnapshot, "CropId"));
            string plotPhase = Convert.ToString(V(plotSnapshot, "Phase"));
            if (!string.Equals(crop, CropId, StringComparison.Ordinal))
                throw new InvalidOperationException(
                    "Actual Crop UI did not select " + CropId + ": " + crop);
            if (!string.Equals(plotPhase, "WaitingForMaterials",
                    StringComparison.Ordinal))
            {
                if (Day() >= ExpiryDay)
                    throw new InvalidOperationException(
                        "Crop input authority was not ready within the flood window: " + plotPhase);
                Time.timeScale = 1f;
                return;
            }
            ValidateCropRequirements(plotSnapshot);
            if (Day() != FloodDay)
                throw new InvalidOperationException(
                    "Research/crop preparation missed the Day1 flood window.");
            Time.timeScale = 1f;
            Advance(15);
            return;
        }
        if (phase == 15)
        {
            int day = Day();
            Time.timeScale = 0f;
            if (day != FloodDay)
                throw new InvalidOperationException(
                    "Ordinary calendar skipped the required flood observation: " + day);
            ValidateFloodWindow();
            SpawnExactCropInputs();
            CaptureSowReadiness("after-input-spawn-before-tick");
            Time.timeScale = 1f;
            Advance(16);
            return;
        }
        if (phase == 16)
        {
            if (Day() >= ExpiryDay)
            {
                CaptureSowReadiness("expiry-day-terminal");
                throw new InvalidOperationException(
                    "Actual Sow did not complete during the Day1 flood window.");
            }
            ObserveSow();
            string current = Convert.ToString(V(plotSnapshot, "Phase"));
            if (!string.Equals(current, "Growing", StringComparison.Ordinal))
                return;
            if (!actualSowObserved || !sowUiObserved
                || maximumSowProgress <= 0f)
                throw new InvalidOperationException(
                    "Growing phase lacked same-actor AI/Sow-progress/UI evidence.");
            Advance(17);
            return;
        }
        if (phase == 17)
        {
            int day = Day();
            if (day < ExpiryDay)
                return;
            Time.timeScale = 0f;
            if (day != ExpiryDay)
                throw new InvalidOperationException(
                    "Ordinary calendar skipped natural Day2 expiry: " + day);
            if (RemainingFloodDays() != 0
                || !Mathf.Approximately(DelayMultiplier(), 1f))
                throw new InvalidOperationException(
                    "Flood work delay did not expire at Day2.");
            if (Convert.ToInt32(V(dailyDiagnostic,
                    "LastDailyEvaluationAbsoluteDay")) != ExpiryDay
                || !Convert.ToBoolean(V(dailyDiagnostic,
                    "LastDailyEvaluationSucceeded")))
                throw new InvalidOperationException(
                    "Day2 daily evaluation did not complete successfully.");
            expiredProjectionEvidence = CaptureActorProjection(
                "day2-expired", out expiredActorMultiplier);
            if (expiredActorMultiplier <= 0f
                || float.IsNaN(expiredActorMultiplier)
                || float.IsInfinity(expiredActorMultiplier))
                throw new InvalidOperationException(
                    "Expired same-actor Sow multiplier is not finite positive.");
            plotSnapshot = FindPlotSnapshot(cropRuntime, plotId);
            Finish("PASS", false,
                "Detached prediction, canonical research fixture, actual crop UI selection, normal AI Sow, Day1 flood and Day2 typed expiry all matched.");
        }
    }

    void CreatePredictAndApply()
    {
        object preparation = Resolve("IStartPartyPreparationService");
        Type difficultyType = T("DungeonDifficulty", "DungeonStory.CoreSession");
        Type pressureType = T(
            "DungeonSurvivalPressure", "DungeonStory.CoreSession");
        Type snapshotType = T("PreparedStartPartySnapshot");
        var create = preparation.GetType().GetMethods()
            .Single(method => method.IsPublic
                && string.Equals(method.Name, "TryCreatePreparedSnapshot",
                    StringComparison.Ordinal)
                && method.GetParameters().Length == 5
                && method.GetParameters()[0].ParameterType == difficultyType
                && method.GetParameters()[1].ParameterType == pressureType
                && method.GetParameters()[2].ParameterType == typeof(int)
                && method.GetParameters()[3].ParameterType
                    == snapshotType.MakeByRefType()
                && method.GetParameters()[4].ParameterType
                    == typeof(string).MakeByRefType());
        object[] arguments =
        {
            Enum.Parse(difficultyType, "Normal"),
            Enum.Parse(pressureType, "Standard"),
            RunSeed, null, null
        };
        if (!(bool)create.Invoke(preparation, arguments))
            throw new InvalidOperationException(
                "Seed-5 prepared snapshot failed: " + Convert.ToString(arguments[4]));
        preparedSnapshot = arguments[3];
        if (preparedSnapshot == null || !Convert.ToBoolean(V(preparedSnapshot, "IsValid")))
            throw new InvalidOperationException("Prepared seed-5 snapshot is invalid.");

        string[] participantIds = ((IEnumerable)V(
                preparedSnapshot, "OrderedMembers"))
            .Cast<object>()
            .Select(value => Convert.ToString(V(value, "persistentId")))
            .ToArray();
        if (participantIds.Length != 3
            || participantIds.Any(string.IsNullOrWhiteSpace)
            || participantIds.Distinct(StringComparer.Ordinal).Count() != 3)
            throw new InvalidOperationException(
                "Prepared seed-5 snapshot lacks three unique participant IDs.");
        detachedPrediction = PredictSeasonalSequence(participantIds);

        calendar = Resolve("IGameCalendar", "DungeonStory.CoreSession");
        gameClock = Resolve(
            "DungeonStory.Foundation.IGameClock", "DungeonStory.Foundation");
        seasonalQuery = Resolve("ISeasonalEventQuery");
        delayQuery = Resolve("IContentWorkDelayQuery");
        dailyDiagnostic = Resolve("IV20DailyEvaluationDiagnostic");
        sowWorkType = WorkType("Sow");
        if (Convert.ToBoolean(V(calendar, "IsRunning")) || Day() != FloodDay
            || RemainingFloodDays() != 0
            || !Mathf.Approximately(DelayMultiplier(), 1f)
            || ActiveSeasonalIds().Length != 0)
            throw new InvalidOperationException(
                "WIM043_DAY1_PREAPPLY_NOT_FRESH;refusing calendar reset or event replay.");
        SubscribeEvents();

        object applier = Resolve("IPreparedStartPartyGameplayApplier");
        var apply = applier.GetType().GetMethod("TryApply", new[]
        {
            snapshotType, typeof(string).MakeByRefType()
        });
        object[] applyArguments = { preparedSnapshot, null };
        if (apply == null || !apply.IsPublic
            || !(bool)apply.Invoke(applier, applyArguments))
            throw new InvalidOperationException(
                "Current-scope prepared gameplay apply failed: "
                + Convert.ToString(applyArguments[1]));
        Time.timeScale = 0f;
        initialSurvivalStockEvidence = ValidateInitialSurvivalStock();
        InvokePublicNoArgs(preparation, "Cancel");
        Component currentScope = RuntimeScope();
        if (!ReferenceEquals(currentScope, originalScope)
            || currentScope.gameObject.scene.handle != originalSceneHandle)
            throw new InvalidOperationException(
                "Prepared gameplay apply replaced the protected scope/scene.");
        Time.timeScale = 0f;
    }

    string ValidateInitialSurvivalStock()
    {
        itemRuntime = Resolve("IWorldItemStackRuntime");
        object dropZones = Resolve("IWorldDropZoneQuery");
        var dropoffMethod = dropZones.GetType().GetMethod(
            "TryGetDeliveryDropoff", new[]
            {
                typeof(Vector2Int).MakeByRefType()
            });
        object[] dropoffArguments = { Vector2Int.zero };
        if (dropoffMethod == null || !dropoffMethod.IsPublic
            || !(bool)dropoffMethod.Invoke(dropZones, dropoffArguments))
            throw new InvalidOperationException(
                "Public starter-supply delivery dropoff is unavailable.");
        Vector2Int dropoff = (Vector2Int)dropoffArguments[0];

        Component[] party = ActiveActors();
        if (party.Length != 3)
            throw new InvalidOperationException(
                "Starter-supply access requires the exact three-member party; count="
                + party.Length);
        object[] all = ((IEnumerable)InvokePublicNoArgs(
                itemRuntime, "GetAllStacks"))
            .Cast<object>().Where(value => value != null).ToArray();
        string[] itemIds = { StarterFoodId, StarterWaterId };
        int[] quantities = { StarterFoodQuantity, StarterWaterQuantity };
        List<string> itemRows = new List<string>();
        List<string> accessRows = new List<string>();
        for (int itemIndex = 0; itemIndex < itemIds.Length; itemIndex++)
        {
            string itemId = itemIds[itemIndex];
            int expectedQuantity = quantities[itemIndex];
            object[] stacks = all.Where(value => string.Equals(
                    Convert.ToString(V(value, "ItemId")), itemId,
                    StringComparison.Ordinal))
                .OrderBy(value => Convert.ToString(V(value, "StackId")),
                    StringComparer.Ordinal)
                .ToArray();
            int quantity = stacks.Sum(value => Convert.ToInt32(
                V(value, "Quantity")));
            int available = stacks.Sum(value => Convert.ToInt32(
                V(value, "AvailableQuantity")));
            bool physicalReady = stacks.Length > 0
                && quantity == expectedQuantity
                && available == expectedQuantity
                && stacks.All(value => string.Equals(
                    Convert.ToString(V(value, "State")), "Loose",
                    StringComparison.Ordinal))
                && stacks.All(value => (Vector2Int)V(value, "Position")
                    == dropoff)
                && stacks.All(value => string.IsNullOrWhiteSpace(
                    Convert.ToString(V(value, "DestinationId"))))
                && stacks.All(value => !Convert.ToBoolean(
                    V(value, "Forbidden")));
            itemRows.Add(itemId + ":expected=" + expectedQuantity
                + ";quantity=" + quantity + ";available=" + available
                + ";stacks=" + DescribeStacks(stacks));
            if (!physicalReady)
                throw new InvalidOperationException(
                    "Initial physical survival stock mismatch: "
                    + itemRows[itemRows.Count - 1] + ";dropoff=" + dropoff);

            foreach (Component actor in party)
            {
                object selected = FindPublicAvailableStack(actor, itemId);
                string selectedId = Convert.ToString(V(selected, "StackId"));
                Vector2Int selectedPosition =
                    (Vector2Int)V(selected, "Position");
                if (!stacks.Any(value => string.Equals(
                        Convert.ToString(V(value, "StackId")), selectedId,
                        StringComparison.Ordinal)))
                    throw new InvalidOperationException(
                        "Initial public available-stock mismatch: actor="
                        + ActorId(actor) + ";item=" + itemId
                        + ";selected=" + selectedId + "@" + selectedPosition);
                accessRows.Add(ActorId(actor) + ":" + itemId + "="
                    + selectedId + "@" + selectedPosition
                    + "/available-query;pathReachability=NOT_CHECKED");
            }
        }
        return "dropoff=" + dropoff + ";items=["
            + string.Join("|", itemRows) + "];publicAccess=["
            + string.Join("|", accessRows) + "]";
    }

    object FindPublicAvailableStack(Component actor, string itemId)
    {
        var method = itemRuntime.GetType().GetMethod(
            "TryFindBestAvailableStack", new[]
            {
                typeof(Vector2Int), typeof(Func<string, int>),
                T("WorldItemStackSnapshot").MakeByRefType()
            });
        if (method == null || !method.IsPublic)
            throw new InvalidOperationException(
                "Public exact available-stock query is unavailable.");
        Func<string, int> rank = candidate => string.Equals(
                candidate, itemId, StringComparison.Ordinal)
            ? 0 : int.MaxValue;
        object[] arguments =
        {
            (Vector2Int)InvokePublicNoArgs(actor, "GetNowXY"), rank, null
        };
        if (!(bool)method.Invoke(itemRuntime, arguments)
            || arguments[2] == null
            || !string.Equals(Convert.ToString(V(arguments[2], "ItemId")),
                itemId, StringComparison.Ordinal))
            throw new InvalidOperationException(
                "Public available-stock query did not select " + itemId
                + " for " + ActorId(actor) + ".");
        return arguments[2];
    }

    static string DescribeStacks(IEnumerable<object> stacks)
    {
        return string.Join(",", stacks.Select(value =>
            Convert.ToString(V(value, "ItemId")) + "@"
            + Convert.ToString(V(value, "StackId"))
            + ":q" + Convert.ToInt32(V(value, "Quantity"))
            + ":r" + Convert.ToInt32(V(value, "ReservedQuantity"))
            + ":a" + Convert.ToInt32(V(value, "AvailableQuantity"))
            + ":state=" + Convert.ToString(V(value, "State"))
            + ":pos=" + Convert.ToString(V(value, "Position"))
            + ":dest=" + Convert.ToString(V(value, "DestinationId"))
            + ":forbidden=" + Convert.ToBoolean(V(value, "Forbidden"))));
    }

    string PredictSeasonalSequence(string[] participantIds)
    {
        object root = Activator.CreateInstance(T(
            "DungeonRuntimeAggregateRootStore", "DungeonStory.Foundation"));
        object catalog = Resolve("V20StoryContentCatalog");
        object candidate = Activator.CreateInstance(
            T("V20CampaignRuntime"), new[] { root, catalog });
        Type contextType = T("V20DailyEventContext");
        Type seasonType = T("Season", "DungeonStory.CoreSession");
        var evaluate = candidate.GetType().GetMethod(
            "EvaluateDaily", new[] { contextType });
        if (evaluate == null || !evaluate.IsPublic)
            throw new InvalidOperationException(
                "Detached public EvaluateDaily is unavailable.");
        List<string> days = new List<string>();
        for (int day = FloodDay; day <= ExpiryDay; day++)
        {
            object context = Activator.CreateInstance(contextType);
            SetV(context, "AbsoluteDay", day);
            SetV(context, "RunSeed", RunSeed);
            SetV(context, "Season", Enum.Parse(seasonType, "Spring"));
            SetV(context, "Generation", 0);
            IList ids = (IList)V(context, "ParticipantCharacterIds");
            foreach (string participantId in participantIds)
                ids.Add(participantId);
            object[] results = ((IEnumerable)evaluate.Invoke(
                    candidate, new[] { context }))
                .Cast<object>().ToArray();
            object[] active = ((IEnumerable)V(
                    candidate, "ActiveSeasonalEvents"))
                .Cast<object>().ToArray();
            string activeId = active.Length == 1
                ? Convert.ToString(V(active[0], "definitionId"))
                : "<count:" + active.Length + ">";
            days.Add(day + ":" + activeId + ":" + string.Join(",",
                results.Select(value => Convert.ToString(V(value, "DefinitionId"))
                    + "/" + Convert.ToString(V(value, "ResolutionId")))));
            string expected = FloodEventId;
            if (active.Length != 1
                || !string.Equals(activeId, expected, StringComparison.Ordinal))
                throw new InvalidOperationException(
                    "Detached seed-5 seasonal prediction mismatch at Day "
                    + day + ": " + string.Join("|", days));
            if (day == FloodDay && !results.Any(value =>
                    string.Equals(Convert.ToString(V(value, "DefinitionId")),
                        FloodEventId, StringComparison.Ordinal)
                    && string.Equals(Convert.ToString(V(value, "ResolutionId")),
                        "started", StringComparison.Ordinal)))
                throw new InvalidOperationException(
                    "Detached Day1 prediction did not start the flood.");
        }
        return string.Join("|", days);
    }

    void SetupGameplay(Component[] loadedActors)
    {
        Time.timeScale = 0f;
        Component currentScope = RuntimeScope();
        if (!ReferenceEquals(currentScope, originalScope)
            || currentScope.gameObject.scene.handle != originalSceneHandle)
            throw new InvalidOperationException(
                "Gameplay setup no longer owns the original protected scope/scene.");
        actors.Clear();
        actors.AddRange(loadedActors);
        object runSeed = Resolve("IRunSeedProvider");
        if (Convert.ToInt32(V(runSeed, "RunSeed")) != RunSeed)
            throw new InvalidOperationException("Live run seed is not 5.");
        calendar = Resolve("IGameCalendar", "DungeonStory.CoreSession");
        gameClock = Resolve(
            "DungeonStory.Foundation.IGameClock", "DungeonStory.Foundation");
        if (Day() != FloodDay || !Convert.ToBoolean(V(calendar, "IsRunning"))
            || !string.Equals(Convert.ToString(V(calendar, "Season")),
                "Spring", StringComparison.Ordinal))
            throw new InvalidOperationException(
                "Live calendar did not begin on Spring Day 1.");
        seasonalQuery = Resolve("ISeasonalEventQuery");
        delayQuery = Resolve("IContentWorkDelayQuery");
        dailyDiagnostic = Resolve("IV20DailyEvaluationDiagnostic");
        if (!ActiveSeasonalIds().SequenceEqual(
                new[] { FloodEventId }, StringComparer.Ordinal))
            throw new InvalidOperationException(
                "Live Day1 seasonal producer did not select the predicted flood: "
                + string.Join(",", ActiveSeasonalIds()));
        ResolveLiveGrid();
        SetupResearchAuthority();
        CompleteFixtureResearch();
    }

    void ResolveLiveGrid()
    {

        Type gridType = T("Grid", "DungeonStory.Grid");
        var tryGetGrid = world.GetType().GetMethod(

    }

    void PlaceInitialP23(string expectedCropId = null)
    {
        ResolveLiveGrid();
        object controller = V(
            Resolve("IDungeonGridBuildingControllerProvider"), "Controller");
        object archetypes = Resolve("IRuntimeBuildingArchetypeCatalog");
        var require = archetypes.GetType().GetMethod(
            "RequireDefinition", new[] { typeof(int) });
        object definition = require.Invoke(archetypes,
            new object[] { PlotDefinitionId });
        var footprintMethod = definition.GetType().GetMethod(
            "GetGridPosList", new[] { typeof(Vector2Int) });
        if (footprintMethod == null || !footprintMethod.IsPublic)
            throw new InvalidOperationException(
                "Public P23 footprint definition is unavailable.");
        object placementLayer = V(V(definition, "Placement"), "Layer");
        Type gridCellType = T("GridCell", "DungeonStory.Grid");
        var canBuildInArea = gridCellType.GetMethod("CanBuildInArea");
        var canOccupy = gridCellType.GetMethod(
            "CanOccupy", new[] { placementLayer.GetType() });
        var getOccupant = gridCellType.GetMethod(
            "GetOccupant", new[] { placementLayer.GetType() });
        if (canBuildInArea == null || !canBuildInArea.IsPublic
            || canOccupy == null || !canOccupy.IsPublic
            || getOccupant == null || !getOccupant.IsPublic)
            throw new InvalidOperationException(
                "Public P23 initial-placement cell contract is unavailable.");
        var getCell = liveGrid.GetType().GetMethod(
            "GetGridCell", new[] { typeof(Vector2Int) });
        if (getCell == null || !getCell.IsPublic)
            throw new InvalidOperationException("Public Grid.GetGridCell is unavailable.");
        cropRuntime = Resolve("ICropPlotRuntime");
        object[] existingPlots = ((IEnumerable)V(cropRuntime, "Plots"))
            .Cast<object>()
            .Where(value => Convert.ToInt32(V(value, "BuildingId"))
                == PlotDefinitionId)
            .ToArray();
        if (existingPlots.Length > 1)
            throw new InvalidOperationException(
                "Expected zero or one current P23 plot; count="
                + existingPlots.Length);

        Vector2Int anchor;
        string reusedPlotId = string.Empty;
        if (existingPlots.Length == 1)
        {
            plotSnapshot = existingPlots[0];
            anchor = (Vector2Int)V(plotSnapshot, "Position");
            reusedPlotId = IdValue(V(plotSnapshot, "PlotId"));
            if (string.IsNullOrWhiteSpace(reusedPlotId))
                throw new InvalidOperationException(
                    "Existing P23 plot has no stable plot ID.");
            plotPlacementEvidence = "reused;plot=" + reusedPlotId
                + ";layer=" + placementLayer;
        }
        else
        {
        object[] cells = ((IEnumerable)InvokePublicNoArgs(liveGrid, "GetCells"))
            .Cast<object>()
            .OrderBy(value => ((Vector2Int)V(value, "Position")).y)
            .ThenBy(value => ((Vector2Int)V(value, "Position")).x)
            .ToArray();
        Vector2Int? selectedAnchor = null;
        int candidatesChecked = 0;
        int missingCellCandidates = 0;
        int areaRejectedCandidates = 0;
        int occupancyRejectedCandidates = 0;
        foreach (object cell in cells)
        {
            candidatesChecked++;
            Vector2Int candidate = (Vector2Int)V(cell, "Position");
            Vector2Int[] footprint = ((IEnumerable)footprintMethod.Invoke(
                    definition, new object[] { candidate }))
                .Cast<Vector2Int>().ToArray();
            bool missingCell = false;
            bool areaRejected = false;
            bool occupancyRejected = false;
            foreach (Vector2Int position in footprint)
            {
                object targetCell = getCell.Invoke(liveGrid,
                    new object[] { position });
                if (targetCell == null)
                {
                    missingCell = true;
                    continue;
                }
                if (!(bool)canBuildInArea.Invoke(targetCell,
                        new[] { definition }))
                    areaRejected = true;
                if (!(bool)canOccupy.Invoke(targetCell,
                        new[] { placementLayer }))
                    occupancyRejected = true;
            }
            if (missingCell)
                missingCellCandidates++;
            if (areaRejected)
                areaRejectedCandidates++;
            if (occupancyRejected)
                occupancyRejectedCandidates++;
            if (!missingCell && !areaRejected && !occupancyRejected)
            {
                selectedAnchor = candidate;
                break;
            }
        }
        if (!selectedAnchor.HasValue)
            throw new InvalidOperationException(
                "No public P23 initial-placement anchor; candidatesChecked="
                + candidatesChecked
                + ";missingCell=" + missingCellCandidates
                + ";areaRejected=" + areaRejectedCandidates
                + ";layer=" + placementLayer
                + ";occupancyRejected=" + occupancyRejectedCandidates
                + ";initialCommand=not-run:no-candidate.");
        anchor = selectedAnchor.Value;

        Type initialType = T("InitialBuildInfo");
        object initial = Activator.CreateInstance(initialType);
        SetV(initial, "Position", anchor);
        SetV(initial, "Building", definition);
        Array placements = Array.CreateInstance(initialType, 1);
        placements.SetValue(initial, 0);
        var place = controller.GetType().GetMethods()
            .Single(method => method.IsPublic
                && string.Equals(method.Name, "TryPlaceInitialBuildings",
                    StringComparison.Ordinal)
                && method.GetParameters().Length == 2
                && method.GetParameters()[1].ParameterType
                    == typeof(string).MakeByRefType());
        object[] placeArguments = { placements, null };
        bool placed = (bool)place.Invoke(controller, placeArguments);
        plotPlacementEvidence = "created;anchor=" + anchor
            + ";candidatesChecked=" + candidatesChecked
            + ";missingCell=" + missingCellCandidates
            + ";areaRejected=" + areaRejectedCandidates
            + ";layer=" + placementLayer
            + ";occupancyRejected=" + occupancyRejectedCandidates
            + ";initialCommand=" + placed
            + ";message=" + Convert.ToString(placeArguments[1]);
        Debug.Log("WIM043_P23_INITIAL_COMMAND;" + plotPlacementEvidence);
        if (!placed)
            throw new InvalidOperationException(
                "Public P23 initial placement failed: "
                + plotPlacementEvidence);
        }

        Vector2Int[] placedFootprint = ((IEnumerable)footprintMethod.Invoke(
                definition, new object[] { anchor }))
            .Cast<Vector2Int>().ToArray();
        object[] occupants = placedFootprint.Select(position =>
        {
            object cell = getCell.Invoke(liveGrid, new object[] { position });
            return cell == null ? null : getOccupant.Invoke(cell,
                new[] { placementLayer });
        }).ToArray();
        if (occupants.Length == 0 || occupants.Any(value => value == null)
            || occupants.Any(value => !ReferenceEquals(value, occupants[0])))
            throw new InvalidOperationException(
                "P23 footprint did not read back one exact occupant.");
        plotBuilding = occupants[0];
        if (Convert.ToInt32(V(plotBuilding, "id")) != PlotDefinitionId
            || (Vector2Int)V(plotBuilding, "centerPos") != anchor)
            throw new InvalidOperationException(
                "P23 occupant ID/anchor readback mismatch.");
        plotId = IdValue(V(plotBuilding, "PersistentInstanceId"));
        if (string.IsNullOrWhiteSpace(plotId))
            throw new InvalidOperationException("P23 has no persistent building ID.");
        if (!string.IsNullOrWhiteSpace(reusedPlotId)
            && !string.Equals(plotId, reusedPlotId, StringComparison.Ordinal))
            throw new InvalidOperationException(
                "Existing P23 plot/occupant stable-ID mismatch: plot="
                + reusedPlotId + ";occupant=" + plotId);
        plotSnapshot = FindPlotSnapshot(cropRuntime, plotId);
        if (Convert.ToInt32(V(plotSnapshot, "BuildingId")) != PlotDefinitionId
            || (Vector2Int)V(plotSnapshot, "Position") != anchor)
            throw new InvalidOperationException(
                "Crop runtime did not join the exact placed P23.");
        if (!string.Equals(Convert.ToString(V(plotSnapshot, "CropId")),
                expectedCropId ?? CropId, StringComparison.Ordinal))
            throw new InvalidOperationException(
                "Canonical fixture P23 default crop mismatch: "
                + Convert.ToString(V(plotSnapshot, "CropId")));
        plotPlacementEvidence += ";join=plot=" + plotId
            + ";building=" + V(plotSnapshot, "BuildingId")
            + ";position=" + V(plotSnapshot, "Position")
            + ";crop=" + V(plotSnapshot, "CropId");
        Debug.Log("WIM043_P23_INITIAL_PLACEMENT_JOIN;" + plotPlacementEvidence);
    }

    void CompleteFixtureProject(
        System.Reflection.MethodInfo completion,
        object project,
        string expectedId)
    {
        object projectId = V(project, "ProjectId");
        string[] prerequisites = ((IEnumerable)V(project, "Prerequisites"))
            .Cast<object>()
            .Select(value => IdValue(V(value, "ProjectId")))
            .OrderBy(value => value, StringComparer.Ordinal)
            .ToArray();
        if (ProjectBool(projectsState, "IsCompleted", projectId))
            throw new InvalidOperationException(
                "Fixture project was already completed: " + expectedId);
        foreach (string prerequisite in prerequisites)
        {
            object prerequisiteProject = FindProject(researchRuntime, prerequisite);
            if (!ProjectBool(projectsState, "IsCompleted",
                    V(prerequisiteProject, "ProjectId")))
                throw new InvalidOperationException(
                    "Fixture prerequisite is incomplete: project=" + expectedId
                    + ";prerequisite=" + prerequisite);
        }
        object[] arguments = { projectId, string.Empty };
        bool completed = (bool)completion.Invoke(researchRuntime, arguments);
        string failure = Convert.ToString(arguments[1]);
        if (!completed || !string.IsNullOrEmpty(failure)
            || !ProjectBool(projectsState, "IsCompleted", projectId))
            throw new InvalidOperationException(
                "Canonical fixture completion failed: project=" + expectedId
                + ";reason=" + failure);
        Debug.Log("WIM043_FIXTURE_CANONICAL_COMPLETION;project=" + expectedId
            + ";prerequisites=" + string.Join(",", prerequisites)
            + ";normalResearchEvidence=false");
    }

    void ValidateProject(
        object project,
        float requiredWork,
        int maximumResearchers,
        string[] prerequisites,
        string expectedState)
    {
        string[] actualPrerequisites = ((IEnumerable)V(project, "Prerequisites"))
            .Cast<object>()
            .Select(value => IdValue(V(value, "ProjectId")))
            .OrderBy(value => value, StringComparer.Ordinal)
            .ToArray();
        string[] expectedPrerequisites = prerequisites
            .OrderBy(value => value, StringComparer.Ordinal).ToArray();
        if (!Mathf.Approximately(Convert.ToSingle(V(project, "RequiredWork")),
                requiredWork)
            || Convert.ToInt32(V(project, "MaximumResearchers"))
                != maximumResearchers
            || !actualPrerequisites.SequenceEqual(
                expectedPrerequisites, StringComparer.Ordinal)
            || !string.Equals(Convert.ToString(V(project, "BlueprintRule")),
                "None", StringComparison.Ordinal)
            || !string.Equals(ProjectState(researchRuntime, project),
                expectedState, StringComparison.Ordinal))
            throw new InvalidOperationException(
                "Research authoring/readiness mismatch for "
                + IdValue(V(project, "ProjectId")));
    }

    void BuildResearchPriorityTasks()
    {
        ResetPriorityTasks();
        object research = WorkType("Research");
        object haul = WorkType("Haul");
        object restock = WorkType("Restock");
        foreach (Component actor in actors)
        {
            object work;
            if (!TryGetWork(actor, out work))
                continue;
            AddPriorityTask(work, research,
                ReferenceEquals(actor, worker) ? "Priority1" : "Off");
            if (!ReferenceEquals(actor, worker))
            {
                AddPriorityTask(work, sowWorkType, "Off");
                AddPriorityTask(work, haul, "Off");
                AddPriorityTask(work, restock, "Off");
            }
        }
        string researchId = IdValue(research);
        foreach (object definition in WorkDefinitions())
        {
            object workType = V(definition, "WorkTypeId");
            if (!string.Equals(IdValue(workType), researchId,
                    StringComparison.Ordinal))
                AddPriorityTask(workerWork, workType, "Off");
        }
    }

    void BuildSowPriorityTasks()
    {
        ResetPriorityTasks();
        object research = WorkType("Research");
        foreach (Component actor in actors)
        {
            object work;
            if (!TryGetWork(actor, out work))
                continue;
            AddPriorityTask(work, research, "Off");
            AddPriorityTask(work, sowWorkType,
                ReferenceEquals(actor, worker) ? "Priority1" : "Off");
        }
        // The canonical research fixture skips BuildResearchPriorityTasks.
        // Isolate this worker's production work explicitly through the same UI;
        // survival needs and the other actors' normal support work remain live.
        string sowId = IdValue(sowWorkType);
        foreach (object definition in WorkDefinitions())
        {
            object workType = V(definition, "WorkTypeId");
            if (!string.Equals(IdValue(workType), sowId, StringComparison.Ordinal)
                && !string.Equals(IdValue(workType), IdValue(research), StringComparison.Ordinal))
                AddPriorityTask(workerWork, workType, "Off");
        }
    }

    void ResetPriorityTasks()
    {
        taskWorkers.Clear();
        taskWorkTypes.Clear();
        taskSuffixes.Clear();
        taskDesired.Clear();
        taskIndex = 0;
        taskClicks = 0;
    }

    void AddPriorityTask(object work, object workType, string desired)
    {
        string key = ActorId(V(work, "WorkerActor")) + "|" + IdValue(workType);
        for (int index = 0; index < taskWorkers.Count; index++)
        {
            string existing = ActorId(V(taskWorkers[index], "WorkerActor"))
                + "|" + IdValue(taskWorkTypes[index]);
            if (string.Equals(existing, key, StringComparison.Ordinal))
            {
                taskDesired[index] = desired;
                return;
            }
        }
        taskWorkers.Add(work);
        taskWorkTypes.Add(workType);
        taskSuffixes.Add(LegacyWorkName(workType));
        taskDesired.Add(desired);
    }

    bool ProcessPriorityTask()
    {
        if (taskIndex >= taskWorkers.Count)
            return true;
        object work = taskWorkers[taskIndex];
        object workType = taskWorkTypes[taskIndex];
        string current = WorkPriority(work, workType);
        if (string.Equals(current, taskDesired[taskIndex],
                StringComparison.Ordinal))
        {
            taskIndex++;
            taskClicks = 0;
            earliestPlayerFrame = Time.frameCount + 1;
            return false;
        }
        if (taskClicks >= 4)
            throw new InvalidOperationException(
                "Priority UI did not reach " + taskDesired[taskIndex]
                + " for " + IdValue(workType) + ";current=" + current);
        object actor = V(work, "WorkerActor");
        string cell = "Cell_"
            + ((UnityEngine.Object)actor).GetInstanceID()
            + "_" + taskSuffixes[taskIndex];
        Click(FindActiveButton(cell));
        taskClicks++;
        earliestPlayerFrame = Time.frameCount + 1;
        return false;
    }

    bool WaitForResearch(object project, string expectedId)
    {
        object id = V(project, "ProjectId");

        if (Day() >= 4)
            throw new InvalidOperationException(
                "RESEARCH_READINESS_BLOCKER_DAY4;project=" + expectedId
                + ";complete=" + complete + ";progress="
                + Convert.ToString(V(ProjectProgress(projectsState, id), "Progress")));

            return false;

            || count <= 0
            || !researchEventDeltas.TryGetValue(expectedId, out float delta)
            || delta <= 0f)

                + expectedId);

            + ";progress=" + V(ProjectProgress(projectsState, id), "Progress")
            + ";events=" + count + ";delta=" + delta + ";day=" + Day());
        return true;
    }

    void ValidateFloodWindow()
    {
        if (floodEventCount != 1 || floodEventDay != FloodDay || !floodPhysical)
            throw new InvalidOperationException(

                + ";day=" + floodEventDay + ";physical=" + floodPhysical);

                new[] { FloodEventId }, StringComparer.Ordinal))
            throw new InvalidOperationException(
                "Actual Day1 active seasonal event is not the flood.");
        if (Convert.ToInt32(V(dailyDiagnostic,
                "LastDailyEvaluationAbsoluteDay")) != FloodDay
            || !Convert.ToBoolean(V(dailyDiagnostic,
                "LastDailyEvaluationSucceeded")))
            throw new InvalidOperationException(
                "Day1 daily evaluation did not succeed.");
        if (RemainingFloodDays() != 1
            || !Mathf.Approximately(DelayMultiplier(), 0.8f))
            throw new InvalidOperationException(
                "Day1 flood delay authority is not 1 day / 0.8x.");
        floodProjectionEvidence = CaptureActorProjection(
            "day1-flood", out floodActorMultiplier);
        if (floodActorMultiplier <= 0f
            || float.IsNaN(floodActorMultiplier)
            || float.IsInfinity(floodActorMultiplier))
            throw new InvalidOperationException(
                "Day1 same-actor Sow context multiplier is not finite positive.");
        floodWindowEvidence = "active=" + string.Join(",", ActiveSeasonalIds())
            + ";remaining=1;typed=0.8;preProjection=["
            + preFloodProjectionEvidence + "];floodProjection=["
            + floodProjectionEvidence + "]";
    }

    void SpawnExactCropInputs()
    {
        plotSnapshot = FindPlotSnapshot(cropRuntime, plotId);
        ValidateCropRequirements(plotSnapshot);
        Vector2Int position = (Vector2Int)V(plotSnapshot, "Position");
        object itemState = Enum.Parse(
            T("WorldItemStackState", "DungeonStory.Items"),
            "FacilityBuffer");
        object economy = Resolve(
            "IResourceEconomyContentCatalog", "DungeonStory.Economy");
        var tryCrop = economy.GetType().GetMethod(
            "TryGetCrop", new[]
            {
                typeof(string),
                T("CropDefinitionSO", "DungeonStory.Economy").MakeByRefType()
            });
        object[] cropArguments = { CropId, null };
        if (tryCrop == null || !tryCrop.IsPublic
            || !(bool)tryCrop.Invoke(economy, cropArguments))
            throw new InvalidOperationException(
                "Authored twilight-grain crop is unavailable.");
        object crop = cropArguments[1];
        if (!string.Equals(Convert.ToString(V(crop, "SeedItemId")),
                SeedItemId, StringComparison.Ordinal))
            throw new InvalidOperationException(
                "Twilight-grain seed item authority mismatch.");

        object seedState = Activator.CreateInstance(
            T("SeedLotState", "DungeonStory.Economy"));
        SetV(seedState, "cropId", CropId);
        SetV(seedState, "cultivarGenomeId",
            Convert.ToString(V(V(crop, "BaseGenome"), "GenomeId")));
        SetV(seedState, "generation", 0);
        SetV(seedState, "pathogenLoad", 0f);
        Type codec = T("SeedLotItemStateCodec", "DungeonStory.Economy");
        var encode = codec.GetMethod("Encode", new[] { seedState.GetType() });
        object component = encode.Invoke(null, new[] { seedState });
        Type componentType = T(
            "ItemInstanceComponentSaveData", "DungeonStory.Items");
        Array components = Array.CreateInstance(componentType, 1);
        components.SetValue(component, 0);

        object transfers = Resolve("IItemTransferService");
        var spawnSeed = transfers.GetType().GetMethods()
            .Single(method => method.IsPublic
                && string.Equals(method.Name, "TrySpawnItemWithComponents",
                    StringComparison.Ordinal)
                && method.GetParameters().Length == 7);
        object[] seedArguments =
        {
            SeedItemId, 1, position, itemState,
            materialDestinationId, components, 0
        };
        bool seedOk = (bool)spawnSeed.Invoke(transfers, seedArguments);
        int seedCount = Convert.ToInt32(seedArguments[6]);

        object items = Resolve("IWorldItemStackRuntime");
        Type stateType = T("WorldItemStackState", "DungeonStory.Items");
        int clean = SpawnPlain(items, stateType, "resource:clean-water",
            position, itemState);
        int nitrate = SpawnPlain(items, stateType, "supply:nitrate-fertilizer",
            position, itemState);
        if (!seedOk || seedCount != 1 || clean != 1 || nitrate != 1)
            throw new InvalidOperationException(
                "Exact crop physical input spawn mismatch: seed=" + seedCount
                + ";water=" + clean + ";nitrate=" + nitrate);
        inputEvidence = SeedItemId + "x1|resource:clean-waterx1|"
            + "supply:nitrate-fertilizerx1;destination="
            + materialDestinationId + ";position=" + position;
    }

    int SpawnPlain(
        object items,
        Type stateType,
        string itemId,
        Vector2Int position,
        object state)
    {
        var method = items.GetType().GetMethod("SpawnItemAt", new[]
        {
            typeof(string), typeof(int), typeof(Vector2Int), stateType,
            typeof(string), typeof(int).MakeByRefType()
        });
        if (method == null || !method.IsPublic)
            throw new InvalidOperationException(
                "Public exact SpawnItemAt overload is unavailable.");
        object[] arguments =
        {
            itemId, 1, position, state, materialDestinationId, 0
        };
        bool ok = (bool)method.Invoke(items, arguments);
        int spawned = Convert.ToInt32(arguments[5]);
        if (!ok)
            throw new InvalidOperationException(
                "Physical crop input spawn rejected: " + itemId);
        return spawned;
    }

    void ObserveSow()
    {
        float now = Convert.ToSingle(V(gameClock, "Time"));
        if (lastSowReadinessGameTime < 0f || now - lastSowReadinessGameTime >= 15f)
            CaptureSowReadiness("day1-observation");
        plotSnapshot = FindPlotSnapshot(cropRuntime, plotId);
        float progress = Convert.ToSingle(V(plotSnapshot, "SowProgress"));
        if (progress > maximumSowProgress)
            maximumSowProgress = progress;
        string assigned = IdValue(V(workerWork, "AssignedWorkTypeId"));
        bool working = Convert.ToBoolean(V(workerWork, "isWorking"));
        object brain = V(worker, "Brain");
        string action = brain == null ? "<no-brain>"
            : Convert.ToString(V(brain, "CurrentActionDebugLabel"))
                + "/" + Convert.ToString(V(brain, "CurrentActionPhase"))
                + "/" + Convert.ToString(V(brain, "CurrentActionPhaseDetail"));
        if (progress > 0f && working
            && string.Equals(assigned, IdValue(sowWorkType),
                StringComparison.Ordinal))
        {
            actualSowObserved = true;
            sowActionEvidence = "actor=" + expectedActorId
                + ";working=true;assigned=" + assigned
                + ";progress=" + progress + ";action=" + action;
            if (!sowUiObserved
                && string.Equals(Convert.ToString(V(plotSnapshot, "Phase")),
                    "Sowing", StringComparison.Ordinal))

                DisplayPlot();

                    .Where(value => value != null
                        && value.gameObject.scene.isLoaded
                        && value.gameObject.activeInHierarchy
                        && string.Equals(value.name, "CropProgress",
                            StringComparison.Ordinal))
                    .ToArray();
                if (roots.Length != 1)
                    throw new InvalidOperationException(
                        "Expected one active CropProgress; count=" + roots.Length);
                Type textType = T("TMPro.TMP_Text", "Unity.TextMeshPro");
                string[] labels = roots[0].GetComponentsInChildren(textType, true)
                    .Cast<Component>()
                    .Select(value => Convert.ToString(V(value, "text")))
                    .Where(value => !string.IsNullOrWhiteSpace(value))
                    .ToArray();
                if (labels.Length != 1
                    || labels[0].IndexOf("파종 중", StringComparison.Ordinal) < 0)
                    throw new InvalidOperationException(
                        "CropProgress did not render actual Sowing: "
                        + string.Join("|", labels));
                sowUiObserved = true;
                sowUiEvidence = labels[0];
                Time.timeScale = 1f;
            }

    IDisposable SubscribeGeneric(
        object eventBus,
        Type eventType,
        string callbackName)
    {
        var subscribe = eventBus.GetType().GetMethods()
            .Single(method => method.IsPublic
                && method.IsGenericMethodDefinition
                && string.Equals(method.Name, "Subscribe",
                    StringComparison.Ordinal)
                && method.GetParameters().Length == 1);
        var callback = GetType().GetMethod(callbackName);
        if (subscribe == null || callback == null || !callback.IsPublic
            || !callback.IsGenericMethodDefinition)
            throw new InvalidOperationException(
                "Public event subscription/callback is unavailable: "
                + callbackName);
        var closed = callback.MakeGenericMethod(eventType);
        Delegate action = Delegate.CreateDelegate(
            typeof(Action<>).MakeGenericType(eventType), this, closed);
        IDisposable subscription = subscribe.MakeGenericMethod(eventType)
            .Invoke(eventBus, new object[] { action }) as IDisposable;
        return subscription ?? throw new InvalidOperationException(
            "Event subscription returned null: " + callbackName);
    }

    public void OnResearchEvent<TEvent>(TEvent value)
    {
        try
        {
            object evt = value;
            string projectId = Convert.ToString(V(evt, "ProjectId"));
            string actorId = IdValue(V(evt, "Researcher"));
            float approved = Convert.ToSingle(V(evt, "ApprovedWork"));
            float delta = Convert.ToSingle(V(evt, "ProgressDelta"));
            int day = Convert.ToInt32(V(evt, "AbsoluteDay"));
            researchEvents.Add("day=" + day + ";project=" + projectId
                + ";actor=" + actorId + ";approved=" + approved
                + ";delta=" + delta);
            if (projectId != GatheringId && projectId != FieldId)
                return;
            if (!string.Equals(actorId, expectedActorId,
                    StringComparison.Ordinal)
                || approved <= 0f || delta <= 0f)
            {
                callbackFailure = true;
                callbackFailureDetail =
                    "Research target had nonmatching contributor/delta: "
                    + researchEvents[researchEvents.Count - 1];
                return;
            }
            researchEventCounts[projectId] =
                researchEventCounts.TryGetValue(projectId, out int count)
                    ? count + 1 : 1;
            researchEventDeltas[projectId] =
                researchEventDeltas.TryGetValue(projectId, out float sum)
                    ? sum + delta : delta;
        }
        catch (Exception exception)
        {
            callbackFailure = true;
            callbackFailureDetail = "Research callback failed: "
                + exception.GetType().Name + ": " + exception.Message;
        }
    }

                actor, "CurrentLifecycleState"))

    static string ReadNeed(Component actor, string conditionName)
    {
        object stats = V(actor, "Stats");
        Type conditionType = T(
            "CharacterCondition", "DungeonStory.Characters");
        var method = stats.GetType().GetMethod(
            "TryGetConditionValue", new[]
            {
                conditionType, typeof(float).MakeByRefType()
            });
        if (method == null || !method.IsPublic)
            return "<query-missing>";
        object[] arguments =
        {
            Enum.Parse(conditionType, conditionName), 0f
        };
        return (bool)method.Invoke(stats, arguments)
            ? Convert.ToSingle(arguments[1]).ToString("0.###")
            : "<missing>";
    }
